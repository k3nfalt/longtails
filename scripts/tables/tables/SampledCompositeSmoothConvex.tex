\caption{Comparison for \href{https://github.com/k3nfalt/longtails/blob/main/LongTails/Setups/SampledCompositeSmoothConvex.lean}{Sampled\allowbreak{}Composite\allowbreak{}Smooth\allowbreak{}Convex}.}\label{tab:SampledCompositeSmoothConvex}
\parbox{\linewidth}{\footnotesize Minimize $\Phi=f+\psi$ on a closed convex set $Q$ in a separable real Hilbert space, where $f=\int F_\xi\,dp$, every $F_\xi$ is convex and globally $L$-smooth, and $\psi:E\to\mathbb{R}$ is lower semicontinuous and convex on $Q$. Samples are IID; sample gradients are jointly measurable, losses integrable, and exact proximal minimizers are available. For some minimizer $x_\star\in Q$, $x_0\in Q$, $\|x_0-x_\star\|\le R$, $\Phi(x_0)-\Phi(x_\star)\le\Delta$, and $\int\|\nabla F_\xi(x_\star)-\int\nabla F_\zeta(x_\star)\,dp(\zeta)\|^2dp(\xi)\le\sigma^2$, with the corresponding integrability conditions. Here $L,R,\varepsilon>0$ and $\Delta,\sigma\ge0$. The setup restricts the algorithm to proximal SGD with linear decay and terminal output; the target is $\int[\Phi(x_T(\omega))-\Phi(x_\star)]\,dP(\omega)\le\varepsilon$.}\par\smallskip

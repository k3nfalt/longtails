\caption{Stochastic nonlinear inequalities: \href{https://github.com/k3nfalt/longtails/blob/main/LongTails/Setups/StochasticSurjectiveInequalities.lean}{StochasticSurjectiveInequalities}.}\label{tab:StochasticSurjectiveInequalities}
\parbox{\linewidth}{\footnotesize Minimize $f(x)$ subject to $c(x)\le0$ in a finite-dimensional real inner-product space, with $m\ge1$, $H\ge1$, $\sigma,\Delta\ge0$, $0<\varepsilon\le1$. Globally, $\|\nabla f\|,\|J\|_F\le H$, $\nabla f,J$ are $H$-Lipschitz, and $\|\lambda\|\le H\|J(x)^\top\lambda\|$ for every $x,\lambda$. The penalty $\Phi=f+4H(H+1)\sum_i[c_i]_+$ satisfies $\Phi(x_0)-\inf\Phi\le\Delta$. Jointly measurable, integrable oracle triples $Z=(g,v,D)$ have mean $(\nabla f,c,J)$, summed variance at most $\sigma^2$, and $\int\|Z(x,\omega)-Z(y,\omega)\|^2\,dp\le H^2\|x-y\|^2$. Samples are IID with law $p$ under probability law $P$; paired evaluations share a sample. Target: output $(x,\lambda\ge0)$ such that, with $P$-probability at least $3/4$, $\|\nabla f(x)+J(x)^\top\lambda\|\le\varepsilon$, $\|[c(x)]_+\|\le\varepsilon$, and $\sum_i|\lambda_i c_i(x)|\le\varepsilon$.}\par\smallskip

\section*{Finite-Sum Convex Strong Growth}
We minimize $F(x)=n^{-1}\sum_{i=1}^n f_i(x)$ on a finite-dimensional real inner product space $E$, where $n\geq1$ is an integer. Each $f_i:E\to\mathbb{R}$ is differentiable and convex with $L$-Lipschitz gradient, $L>0$. Strong growth requires $n^{-1}\sum_{i=1}^n\|\nabla f_i(x)\|^2\leq\rho\|\nabla F(x)\|^2$ for every $x\in E$, with $\rho\geq1$. A given $x^0\in E$ satisfies $\|x^0-x^\star\|\leq R$, where $R\geq0$ and $x^\star$ minimizes $F$. The oracle returns $\nabla f_i(x)$ at a queried pair $(i,x)$. For $\varepsilon>0$, the goal is a measurable output $\bar{x}\in E$ with integrable gap satisfying $\mathbb{E}_{P}[F(\bar{x})-F(x^\star)]\leq\varepsilon$, where $P$ is the law of the algorithm's randomness.

The optimization problem is
\[
\min_{x\in E} F(x).
\]
The norm $\|\cdot\|$ is induced by the inner product on $E$, which is equipped with a measurable structure. The algorithm receives $E$, $n$, $L$, $\rho$, $R$, $\varepsilon$, and $x^0$.

Our assumptions:
\begin{assumption}[Component convexity and smoothness]
Each $f_i$ is Fr\'echet differentiable everywhere on $E$. For every $i\in\{1,\ldots,n\}$, $x,y\in E$, and $t\in[0,1]$,
\[
\begin{aligned}
f_i((1-t)x+ty)&\leq(1-t)f_i(x)+t f_i(y),\\
\|\nabla f_i(x)-\nabla f_i(y)\|&\leq L\|x-y\|.
\end{aligned}
\]
\end{assumption}

\begin{assumption}[Strong growth]
For every $x\in E$,
\[
\frac{1}{n}\sum_{i=1}^n\|\nabla f_i(x)\|^2
\leq \rho\|\nabla F(x)\|^2,
\qquad
\nabla F(x)=\frac{1}{n}\sum_{i=1}^n\nabla f_i(x).
\]
\end{assumption}

\begin{assumption}[Initial distance]
There exists $x^\star\in E$ such that
\[
\begin{aligned}
F(x^\star)&\leq F(x) &&\text{for every }x\in E,\\
\|x^0-x^\star\|&\leq R.
\end{aligned}
\]
\end{assumption}

The algorithm may query any component index $i\in\{1,\ldots,n\}$ at any point $x\in E$; each query returns the exact component gradient. After the allotted component-gradient calls, its output $\bar{x}$ must be measurable and satisfy
\[
\mathbb{E}_{P}\bigl[F(\bar{x})-F(x^\star)\bigr]
\leq\varepsilon.
\]
The gap must be integrable under $P$. Here $x^\star$ is a fixed global minimizer; all global minimizers have the same objective value.

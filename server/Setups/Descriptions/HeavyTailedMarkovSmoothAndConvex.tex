\section*{Convex Minimization with Markov Noise}
We consider minimizing a differentiable convex function $f:E\to\mathbb{R}$ on a separable real Hilbert space $E$. Its gradient is $L$-Lipschitz, where $L>0$, and a given point $x^0\in E$ is within distance $R\geq0$ of a global minimizer $x^\star$. At query $x$, the oracle returns $g(x,Z_k)$, where $(Z_k)_{k\geq0}$ is a stationary, polynomially mixing Markov chain. Under its stationary law, $g$ is unbiased and its centered $\alpha$-moment is at most $\sigma^\alpha$, with $1<\alpha\leq2$ and $\sigma\geq0$. For $\varepsilon>0$, the goal is a measurable output $\bar{x}$ with integrable objective gap and $\mathbb{E}[f(\bar{x})-f(x^\star)]\leq\varepsilon$, averaging over oracle and independent algorithm randomness.

The optimization problem is
\[
\min_{x\in E} f(x).
\]
The norm $\|\cdot\|$ is induced by the inner product on $E$. Let $\Omega$ be a measurable sample space, and let $\tau>0$ and $\beta>0$ be the mixing parameters. The algorithm receives $E$, $\Omega$, $L$, $R$, $\sigma$, $\alpha$, $\tau$, $\beta$, $\varepsilon$, and $x^0$; the objective and the oracle's sampling law are accessed through oracle replies.

Our assumptions:
\begin{assumption}[Convexity and smoothness]
The function $f$ is Fr\'echet differentiable everywhere on $E$. For every $x,y\in E$ and $s\in[0,1]$,
\[
\begin{aligned}
f((1-s)x+sy)&\leq (1-s)f(x)+sf(y),\\
\|\nabla f(x)-\nabla f(y)\|&\leq L\|x-y\|.
\end{aligned}
\]
\end{assumption}

\begin{assumption}[Initial distance]
There exists $x^\star\in E$ such that
\[
\begin{aligned}
f(x^\star)&\leq f(x) &&\text{for every }x\in E,\\
\|x^0-x^\star\|&\leq R.
\end{aligned}
\]
\end{assumption}

\begin{assumption}[Stationary Markov sampling and mixing]
The samples form a time-homogeneous Markov chain on $\Omega$ with transition probability kernel $\kappa$ and initial distribution $\pi$, where $\pi$ is a probability measure. It is invariant: for every measurable $A\subseteq\Omega$,
\[
\mathbb{E}_{Z\sim\pi}[\kappa(Z,A)]=\pi(A).
\]
For each integer $t\geq0$, define the total-variation distance using the convention
\[
d_t(z)=\sup_{A\subseteq\Omega\;\mathrm{measurable}}
\bigl|\kappa^t(z,A)-\pi(A)\bigr|,
\]
where $\kappa^t$ is the $t$-step transition kernel and $\kappa^0(z,\cdot)$ is the point mass at $z$. The function $d_t$ is integrable under $\pi$ and satisfies
\[
\mathbb{E}_{Z\sim\pi}[d_t(Z)]
\leq \left(1+\frac{t}{\tau}\right)^{-\beta},
\qquad t=0,1,2,\ldots.
\]
\end{assumption}

\begin{assumption}[Stationary oracle moments]
The map $g:E\times\Omega\to E$ is jointly measurable. For every fixed $x\in E$, $g(x,\cdot)$ is Bochner integrable under $\pi$, the function $\|g(x,\cdot)-\nabla f(x)\|^\alpha$ is integrable under $\pi$, and
\[
\begin{aligned}
\mathbb{E}_{Z\sim\pi}[g(x,Z)]&=\nabla f(x),\\
\mathbb{E}_{Z\sim\pi}\bigl[\|g(x,Z)-\nabla f(x)\|^\alpha\bigr]
&\leq\sigma^\alpha.
\end{aligned}
\]
These moment conditions are taken under the stationary law at each fixed query point.
\end{assumption}

Queries may be made at any point of $E$. The oracle returns $g(x_k,Z_k)$ on call $k$, starting at $k=0$ and using consecutive states of the same chain. The query $x_k$ may depend on previous replies and on algorithm randomness, which is independent of the chain. Any calls whose replies are discarded still count toward the oracle-call budget.

After the allotted calls, the output $\bar{x}\in E$ must be measurable, its objective gap must be integrable, and it must satisfy
\[
\mathbb{E}_{P}\bigl[f(\bar{x})-f(x^\star)\bigr]\leq\varepsilon,
\]
where $P$ is the joint law of the Markov chain and the independent algorithm randomness.

\section*{Radius-Constrained Composite Convex Minimization}
We minimize $F=f+\psi$ over a convex set $Q$ in a real Hilbert space $E$, where $f:E\to\mathbb{R}$ is differentiable and convex and $\psi:E\to\mathbb{R}$ is convex on $Q$. The function $f$ is $(L_0,L_1)$-smooth, with $L_0>0$ and $L_1\geq0$. A given $x^0\in Q$ satisfies $\|x^0-x^\star\|\leq R$ and $\|\nabla f(x^\star)\|\leq G$ for some minimizer $x^\star$, where $R>0$ and $G\geq0$. The oracle returns exact gradients and constrained proximal minimizers. For $\varepsilon>0$, the goal is a feasible output $\bar{x}\in Q$ with $F(\bar{x})-F(x^\star)\leq\varepsilon$ almost surely under $P$, the law of the algorithm's randomness.

Writing $F(x)=f(x)+\psi(x)$, the optimization problem is
\[
\min_{x\in Q} F(x).
\]
The norm $\|\cdot\|$ is induced by the inner product $\langle\cdot,\cdot\rangle$ on $E$. The algorithm receives $E$, $L_0$, $L_1$, $R$, $\varepsilon$, and $x^0$; the bound $G$ may enter only the oracle-call budget.

Our assumptions:
\begin{assumption}[Convexity]
The set $Q\subseteq E$ is convex. For every $t\in[0,1]$,
\[
\begin{aligned}
f((1-t)x+ty)&\leq(1-t)f(x)+tf(y),
&&x,y\in E,\\
\psi((1-t)x+ty)&\leq(1-t)\psi(x)+t\psi(y),
&&x,y\in Q.
\end{aligned}
\]
\end{assumption}

\begin{assumption}[Generalized smoothness]
The function $f$ is Fr\'echet differentiable everywhere on $E$, and $\nabla f:E\to E$ is continuous. For every $x,y\in E$ and $M\in\mathbb{R}$,
\[
\begin{gathered}
\|\nabla f(x+\theta(y-x))\|\leq M
\quad\text{for all }\theta\in[0,1]\\
\Longrightarrow\quad
\|\nabla f(x)-\nabla f(y)\|
\leq (L_0+L_1M)\|x-y\|.
\end{gathered}
\]
\end{assumption}

\begin{assumption}[Exact proximal minimization]
There is a map $T:\mathbb{R}\times E\times E\to E$ such that, for every $a>0$ and $z,v\in E$,
\[
T(a,z,v)\in\operatorname*{arg\,min}_{u\in Q}
\left\{
\frac{\|u-z\|^2}{2a}
+\langle v,u-z\rangle+\psi(u)
\right\}.
\]
\end{assumption}

\begin{assumption}[Initial distance and minimizer gradient]
The given starting point satisfies $x^0\in Q$. There exists $x^\star\in Q$ such that
\[
\begin{aligned}
F(x^\star)&\leq F(x) &&\text{for every }x\in Q,\\
\|x^0-x^\star\|&\leq R,\\
\|\nabla f(x^\star)\|&\leq G.
\end{aligned}
\]
\end{assumption}

The oracle returns $\nabla f(x)$ at $x$. It also returns $T(a,z,v)$ on a proximal query $(a,z,v)$. Gradient queries may use any $x\in E$, and proximal queries may use any $a\in\mathbb{R}$ and $z,v\in E$; the proximal minimization requirement applies only for $a>0$. Both query types count toward a single total oracle-call budget.

After the allotted calls, the output must satisfy
\[
\bar{x}\in Q,
\qquad F(\bar{x})-F(x^\star)\leq\varepsilon
\quad P\text{-a.s.}
\]
Here $P$ is the law of the algorithm's randomness, and $x^\star$ is a fixed minimizer independent of that randomness.

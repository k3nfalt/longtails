\section*{Mean-Squared H\"older Nonconvex Optimization}
We consider finding an approximate stationary point of a differentiable, possibly nonconvex function $f:E\to\mathbb{R}$ on a separable real Hilbert space $E$. Its gradient is $L$-Lipschitz, where $L>0$, and a given point $x^0\in E$ satisfies $f(x^0)-\inf_{x\in E}f(x)\leq\Delta$, where $\Delta\geq0$. At a pair $x,y\in E$, the oracle returns unbiased gradient estimates $g(x,\xi),g(y,\xi)$ using one shared sample $\xi$, with independent samples across calls. Their variance is bounded by $\sigma^2$, and their mean-squared difference by $M\|x-y\|^{2\nu}$, where $\sigma,M\geq0$ and $\nu\in[1/2,1]$. For $\varepsilon>0$, the goal is a measurable output $\bar{x}\in E$ with $\mathbb{E}\|\nabla f(\bar{x})\|\leq\varepsilon$, averaging over oracle and independent algorithm randomness.

The underlying optimization problem is
\[
\inf_{x\in E} f(x).
\]
The norm $\|\cdot\|$ is induced by the inner product on $E$. The algorithm receives $E$, the measurable sample space $\Omega$, $x^0$, and all parameters $L,M,\nu,\sigma,\Delta,\varepsilon$.

Our assumptions:
\begin{assumption}[Smoothness and initial gap]
The function $f$ is Fr\'echet differentiable everywhere on $E$. For every $x,y\in E$,
\[
\begin{aligned}
\|\nabla f(x)-\nabla f(y)\|&\leq L\|x-y\|,\\
f(x^0)-f(x)&\leq\Delta.
\end{aligned}
\]
\end{assumption}

\begin{assumption}[Unbiased stochastic gradients]
There exist a probability measure $p$ on $\Omega$ and a jointly measurable map $g:E\times\Omega\to E$. For every $x\in E$, the random vector $g(x,\xi)$ is Bochner integrable and $\|g(x,\xi)-\nabla f(x)\|^2$ is integrable under $\xi\sim p$, with
\[
\begin{aligned}
\mathbb{E}_{\xi\sim p}[g(x,\xi)]&=\nabla f(x),\\
\mathbb{E}_{\xi\sim p}\bigl[\|g(x,\xi)-\nabla f(x)\|^2\bigr]&\leq\sigma^2.
\end{aligned}
\]
\end{assumption}

\begin{assumption}[Mean-squared H\"older continuity]
For every $x,y\in E$, the random variable $\|g(x,\xi)-g(y,\xi)\|^2$ is integrable under $\xi\sim p$, and
\[
\mathbb{E}_{\xi\sim p}\bigl[\|g(x,\xi)-g(y,\xi)\|^2\bigr]
\leq M\|x-y\|^{2\nu}.
\]
\end{assumption}

At call $k\geq0$, a query $(x_k,y_k)\in E\times E$ receives the pair
\[
\bigl(g(x_k,\xi_k),g(y_k,\xi_k)\bigr).
\]
The measurable samples $(\xi_k)_{k\geq0}$ are mutually independent with common law $p$, and are independent of the algorithm's randomness. Both evaluations within a pair use the same sample; one returned pair counts as one oracle call. Queries may depend on preceding replies.

After the allotted calls, the output $\bar{x}$ must be measurable, its gradient norm must be integrable, and it must satisfy
\[
\mathbb{E}_{P}\bigl[\|\nabla f(\bar{x})\|\bigr]\leq\varepsilon,
\]
where $P$ is the joint law of the oracle samples and the algorithm's randomness.

\section*{Strong Quasar Minimization with Relative Errors}
We consider minimizing a differentiable function $f:E\to\mathbb{R}$ on a finite-dimensional real Hilbert space $E$. The function has an $L$-Lipschitz gradient, with $L>0$, and is strongly quasar-convex with parameters $\mu>0$ and $\gamma\in(0,1]$ relative to a global minimizer $x^\star$. A given point $x^0\in E$ satisfies $f(x^0)-f(x^\star)\leq\Delta$, where $\Delta\geq0$. The oracle returns a fixed measurable vector $g(x)$ satisfying $\|g(x)-\nabla f(x)\|\leq\delta\|\nabla f(x)\|$, with $\delta\in[0,1)$. For $\varepsilon>0$, the goal is an output $\bar{x}\in E$ with integrable objective gap and $\mathbb{E}_{P}[f(\bar{x})-f(x^\star)]\leq\varepsilon$, where $P$ is the law of the algorithm's randomness.

The optimization problem is
\[
\min_{x\in E} f(x).
\]
The norm $\|\cdot\|$ is induced by the inner product $\langle\cdot,\cdot\rangle$ on $E$. The algorithm receives $E$, $L$, $x^0$, $\mu$, $\gamma$, $\delta$, $\Delta$, and $\varepsilon$.

Our assumptions:
\begin{assumption}[Smoothness]
The function $f$ is Fr\'echet differentiable everywhere on $E$, and
\[
\|\nabla f(x)-\nabla f(y)\|\leq L\|x-y\|,
\qquad x,y\in E.
\]
\end{assumption}

\begin{assumption}[Strong quasar-convexity and initial gap]
There exists $x^\star\in E$ such that
\[
\begin{aligned}
f(x^\star)&\leq f(x), \qquad x\in E,\\
f(x^0)-f(x^\star)&\leq\Delta.
\end{aligned}
\]
Relative to this same minimizer, strong quasar-convexity means that, for every $x\in E$,
\[
f(x^\star)\geq f(x)
+\frac{1}{\gamma}\langle\nabla f(x),x^\star-x\rangle
+\frac{\mu}{2}\|x-x^\star\|^2.
\]
\end{assumption}

\begin{assumption}[Relative gradient error]
The oracle returns $g(x)$ at any query point $x\in E$, where $g:E\to E$ is a fixed Borel-measurable map satisfying
\[
\|g(x)-\nabla f(x)\|\leq\delta\|\nabla f(x)\|,
\qquad x\in E.
\]
\end{assumption}

After the allotted oracle calls, the output must have an integrable objective gap under $P$ and satisfy
\[
\mathbb{E}_{P}\bigl[f(\bar{x})-f(x^\star)\bigr]\leq\varepsilon.
\]

\section*{Nonconvex Optimization with Curvature-Aligned Noise}
We consider minimizing a differentiable, possibly nonconvex function $f:E\to\mathbb{R}$ on a finite-dimensional real inner product space $E$. The objective admits global quadratic upper models with bounded curvature along its gradient and a bounded initial gap to a lower bound. The oracle returns an unbiased stochastic gradient at each query, using independent, identically distributed samples. Its noise has bounded variance and bounded expected curvature relative to its second moment. For $\varepsilon>0$, the goal is a measurable output $\bar{x}\in E$ with $\mathbb{E}\|\nabla f(\bar{x})\|\leq\varepsilon$, where expectation includes oracle and independent algorithm randomness.

The optimization problem is
\[
\inf_{x\in E} f(x).
\]
Write $\langle\cdot,\cdot\rangle$ for the inner product and $\|\cdot\|$ for its induced norm. Fix a starting point $x^0\in E$ and parameters
\[
L_g>0,\qquad L_\xi\geq0,\qquad \sigma\geq0,\qquad
\Delta\geq0,\qquad \varepsilon>0.
\]
The algorithm receives $E$, the measurable sample space $\Omega$, $x^0$, and all five parameters; the objective and oracle distribution are accessed through oracle replies.

Our assumptions:
\begin{assumption}[Quadratic upper models]
The function $f$ is Fr\'echet differentiable everywhere. There exists a family of continuous linear maps $Q(x):E\to E$ such that, for every $x,u\in E$,
\[
\begin{aligned}
f(x+u)&\leq f(x)+\langle\nabla f(x),u\rangle
+\tfrac12\langle Q(x)u,u\rangle,\\
\langle Q(x)\nabla f(x),\nabla f(x)\rangle
&\leq L_g\|\nabla f(x)\|^2.
\end{aligned}
\]
\end{assumption}

\begin{assumption}[Initial objective gap]
There exists $f_{\mathrm{low}}\in\mathbb{R}$ such that
\[
\begin{aligned}
f(x)&\geq f_{\mathrm{low}} &&\text{for every }x\in E,\\
f(x^0)-f_{\mathrm{low}}&\leq\Delta.
\end{aligned}
\]
\end{assumption}

\begin{assumption}[Unbiased curvature-aligned noise]
There exist a probability measure $\nu$ on $\Omega$ and a jointly measurable map $g:E\times\Omega\to E$. Define the noise by
\[
\zeta(x,\omega)=g(x,\omega)-\nabla f(x).
\]
For every fixed $x\in E$, the map $g(x,\cdot)$ is Bochner integrable under $\nu$, and both $\|\zeta(x,\cdot)\|^2$ and $\langle Q(x)\zeta(x,\cdot),\zeta(x,\cdot)\rangle$ are integrable under $\nu$. For $\omega\sim\nu$, they satisfy
\[
\begin{aligned}
\mathbb{E}_{\omega\sim\nu}[g(x,\omega)]&=\nabla f(x),\\
\mathbb{E}_{\omega\sim\nu}\|\zeta(x,\omega)\|^2&\leq\sigma^2,\\
\mathbb{E}_{\omega\sim\nu}
\langle Q(x)\zeta(x,\omega),\zeta(x,\omega)\rangle
&\leq L_\xi\mathbb{E}_{\omega\sim\nu}\|\zeta(x,\omega)\|^2.
\end{aligned}
\]
At call $k\geq0$, a query $x\in E$ returns $g(x,\xi_k)$, where the measurable sample variables $(\xi_k)_{k\geq0}$ are mutually independent and each has law $\nu$. The algorithm's randomness is independent of the entire sample sequence.
\end{assumption}

Queries may be made at any point of $E$. After the allotted oracle calls, the output $\bar{x}$ must be measurable, its gradient norm must be integrable, and it must satisfy
\[
\mathbb{E}_{P}\bigl[\|\nabla f(\bar{x})\|\bigr]\leq\varepsilon,
\]
where $P$ is the joint law of the oracle samples and the algorithm's randomness.

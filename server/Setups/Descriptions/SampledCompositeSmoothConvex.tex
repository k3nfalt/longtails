\section*{Sampled Smooth Composite Convex Optimization}
We consider minimizing $F(x)=\int_\Xi \ell(\xi,x)\,p(d\xi)+\psi(x)$ over a closed convex set $Q$ in a separable real Hilbert space $E$, where $p$ is a probability law on a measurable sample space $\Xi$. Each sampled loss $\ell(\xi,\cdot)$ is convex with $L$-Lipschitz gradient, $L>0$, and $\psi:E\to\mathbb{R}$ is lower semicontinuous and convex on $Q$. A minimizer $x^\star$ satisfies prescribed initial distance, objective-gap, and gradient-variance bounds. One oracle supplies independent sampled gradients or exact proximal minimizers. The required last iterate $\bar{x}$ is measurable, has integrable objective gap, and satisfies $\mathbb{E}[F(\bar{x})-F(x^\star)]\leq\varepsilon$, where $\varepsilon>0$ is the target accuracy.

The problem is
\[
\min_{x\in Q} F(x),
\qquad
F(x)=f(x)+\psi(x),
\qquad
f(x)=\int_\Xi \ell(\xi,x)\,p(d\xi).
\]
The space $E$ carries its Borel sigma-algebra and the norm induced by its inner product. The public inputs are $E$, $\Xi$, an initial point $x^0\in E$, and the parameters
\[
L>0,\qquad R>0,\qquad \Delta\geq0,\qquad
\sigma\geq0,\qquad \varepsilon>0.
\]
The instance consists of $p$, the sampled loss $\ell:\Xi\times E\to\mathbb{R}$, a sampled gradient $g:\Xi\times E\to E$, the function $\psi$, the set $Q$, and a map $\operatorname{prox}:\mathbb{R}\times E\times E\to E$. These are accessed through the oracle described below.

\begin{assumption}[Samplewise convexity and smoothness]
The map $(\xi,x)\mapsto g(\xi,x)$ is jointly measurable. For every $\xi\in\Xi$, the function $\ell(\xi,\cdot)$ is differentiable, and for all $x,y\in E$ and $t\in[0,1]$,
\[
\begin{aligned}
\nabla_x\ell(\xi,x)&=g(\xi,x),\\
\ell(\xi,(1-t)x+ty)&\leq(1-t)\ell(\xi,x)+t\ell(\xi,y),\\
\|g(\xi,x)-g(\xi,y)\|&\leq L\|x-y\|.
\end{aligned}
\]
For every $x\in E$, the function $\xi\mapsto\ell(\xi,x)$ is integrable with respect to $p$.
\end{assumption}

\begin{assumption}[Composite term and proximal access]
The set $Q$ is closed and convex, $\psi$ is lower semicontinuous on $E$, and, for every $x,y\in Q$ and $t\in[0,1]$,
\[
\psi((1-t)x+ty)\leq(1-t)\psi(x)+t\psi(y).
\]
For every $a>0$ and $z,v\in E$, the proximal reply satisfies
\[
\operatorname{prox}(a,z,v)\in
\underset{u\in Q}{\operatorname{argmin}}
\left\{\frac{\|u-z\|^2}{2a}
+\langle v,u-z\rangle+\psi(u)\right\}.
\]
The map is defined for all real $a$; its minimization and feasibility guarantees apply when $a>0$.
\end{assumption}

\begin{assumption}[Attainment and initialization]
The initial point belongs to $Q$. There exists $x^\star\in Q$ such that
\[
\begin{aligned}
F(x^\star)&\leq F(x) &&\text{for every }x\in Q,\\
\|x^0-x^\star\|&\leq R,\\
F(x^0)-F(x^\star)&\leq\Delta.
\end{aligned}
\]
\end{assumption}

\begin{assumption}[Gradient variance at the minimizer]
At the same minimizer $x^\star$, the map $\xi\mapsto g(\xi,x^\star)$ is Bochner integrable. With
\[
m^\star=\int_\Xi g(\xi,x^\star)\,p(d\xi),
\]
the function $\xi\mapsto\|g(\xi,x^\star)-m^\star\|^2$ is integrable and satisfies
\[
\int_\Xi\|g(\xi,x^\star)-m^\star\|^2\,p(d\xi)
\leq\sigma^2.
\]
\end{assumption}

\begin{assumption}[Sampling and oracle costs]
There is a probability law $P$ on $\Xi^{\mathbb{N}}$ under which the coordinate samples $(\xi_j)_{j\geq0}$ are measurable, mutually independent, and each has law $p$. The single oracle, with ID $0$, accepts two query types. At call index $j$, a gradient query $x\in E$ returns $g(\xi_j,x)$, whereas a proximal query $(a,z,v)\in\mathbb{R}\times E\times E$ returns $\operatorname{prox}(a,z,v)$ deterministically. Each query costs one call to this same oracle. Query points and proximal arguments may lie anywhere in their stated domains. Algorithm randomness is independent of the oracle randomness.
\end{assumption}

The allowed algorithm family is proximal stochastic gradient descent with linear step-size decay. It is selected using only the public inputs and is specified by $\eta\geq0$ and $T\in\{0,1,2,\ldots\}$. Starting from $x^0$, it alternates a gradient call and a proximal call according to
\[
\begin{aligned}
a_k&=\eta\frac{T-k}{T},\\
v_k&=g(\xi_{2k},x^k),\\
x^{k+1}&=\operatorname{prox}(a_k,x^k,v_k),
\qquad k=0,1,2,\ldots.
\end{aligned}
\]
Here the quotient is interpreted as zero when $T=0$. Each complete update costs two oracle calls; the sample at index $2k+1$ does not affect the proximal reply. The parameter $T$ controls the step-size schedule, while an externally specified budget $K\in\{0,1,2,\ldots\}$ controls the number of calls. After $K$ calls, the output is the last completed iterate
\[
\bar{x}=x^{\lfloor K/2\rfloor}.
\]
An unmatched final gradient call is counted and leaves this output unchanged. The budget may depend on the full public parameter tuple.

Writing $\mathbb{P}$ for the joint probability law of algorithm and oracle randomness, success requires that $\bar{x}$ be measurable and that, for a minimizer $x^\star\in Q$, the function $\omega\mapsto F(\bar{x}(\omega))-F(x^\star)$ be integrable and satisfy
\[
\int \bigl(F(\bar{x}(\omega))-F(x^\star)\bigr)
\,\mathbb{P}(d\omega)\leq\varepsilon.
\]
The minimizer witnessing the initialization and variance conditions may be used here, since all minimizers over $Q$ have the same objective value.

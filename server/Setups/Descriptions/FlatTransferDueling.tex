\section*{Strongly Convex Minimization with Flat Comparisons}
We minimize a continuously differentiable function $f:\mathbb{R}^d\to\mathbb{R}$ over the closed Euclidean unit ball $B$ in dimension $d\geq 1$. On $B$, $f$ is strongly convex and Lipschitz with Lipschitz gradient. At each queried pair $x,y\in B$, the oracle uses a fresh independent sample to return a sign with mean $\rho(f(x)-f(y))$. The unknown transfer function $\rho$ is odd and nondecreasing, with a polynomial lower bound on its slope near zero. For $0<\varepsilon\leq 1/64$, the goal is a feasible output $\bar{x}\in B$ with $\mathbb{E}[f(\bar{x})-f(x^\star)]\leq\varepsilon$, where $x^\star$ minimizes $f$ on $B$ and the expectation includes algorithm and oracle randomness.

Writing $\|\cdot\|$ for the Euclidean norm, the optimization problem is
\[
\min_{x\in B} f(x), \qquad B=\{x\in\mathbb{R}^d:\|x\|\leq 1\}.
\]
The algorithm receives $d$, $\varepsilon$, the objective constants specified below, and the transfer parameters $r=1$, $k=2j$, $c=(2j+1)/8$, and $M=(2j+1)/4$, where $j$ is a nonnegative integer.

Our assumptions:
\begin{assumption}[Objective regularity and curvature]
The function $f$ belongs to $C^1(\mathbb{R}^d)$. For every $x,y\in B$,
\[
\begin{aligned}
|f(x)-f(y)| &\leq 4\|x-y\|,\\
\|\nabla f(x)-\nabla f(y)\| &\leq 2\|x-y\|,\\
f(y) &\geq f(x)+\langle\nabla f(x),y-x\rangle
             +\tfrac14\|y-x\|^2.
\end{aligned}
\]
\end{assumption}

\begin{assumption}[Transfer function]
The function $\rho:\mathbb{R}\to[-1,1]$ is continuously differentiable, odd, and nondecreasing. Its derivative satisfies
\[
c|t|^k\leq \rho'(t)\leq M, \qquad |t|\leq r,
\]
with $|t|^0=1$ for every $t$.
\end{assumption}

\begin{assumption}[Comparison oracle]
Let $Z_1,Z_2,\ldots$ be independent random variables with common law $\nu$ on a measurable space $\mathcal{Z}$, independent of the algorithm's randomness. On call $i$ at an ordered pair $(x,y)\in B\times B$, the oracle returns $g(x,y,Z_i)$, where $g:B\times B\times\mathcal{Z}\to\{-1,1\}$ is jointly measurable. For every $x,y\in B$,
\[
\mathbb{E}_{Z\sim\nu}[g(x,y,Z)]=\rho(f(x)-f(y)).
\]
\end{assumption}

After the allotted comparisons, the algorithm must return a measurable $B$-valued output $\bar{x}$. Its gap relative to a minimizer $x^\star\in\operatorname*{arg\,min}_{x\in B}f(x)$ must be integrable and satisfy
\[
\mathbb{E}\bigl[f(\bar{x})-f(x^\star)\bigr]\leq\varepsilon.
\]

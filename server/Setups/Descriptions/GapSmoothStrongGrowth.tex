\section*{Gap-Smooth Convex Minimization}
We consider minimizing a differentiable convex function $f:E\to\mathbb{R}$ on a separable real Hilbert space $E$. A local quadratic upper model has curvature controlled by the gap to a global minimizer $x^\star$, and a given starting point has bounded distance and objective gap to that minimizer. The oracle returns independent-sample, unbiased stochastic gradients whose variance is bounded proportionally to the squared gradient norm. For tolerance $\varepsilon>0$ and failure probability $\delta\in(0,1)$, the goal is a measurable output $\bar{x}\in E$ satisfying $P\{f(\bar{x})-f(x^\star)>\varepsilon\}\leq\delta$, where $P$ is the joint law of sampling and independent algorithm randomness.

The optimization problem is
\[
\min_{x\in E} f(x).
\]
Equip $E$ with its Borel sigma-algebra, and write $\|\cdot\|$ for the norm induced by its inner product. Let $\Omega$ be a measurable sample space and $x^0\in E$ the given starting point. The parameters satisfy
\[
\begin{aligned}
&H_0\geq0,\qquad H_1\geq0,\qquad R>0,\qquad \Delta>0,\\
&\rho\geq1,\qquad 0<\varepsilon\leq\Delta,\qquad 0<\delta<1.
\end{aligned}
\]
The algorithm receives $E$, $\Omega$, $x^0$, and all seven parameters $H_0,H_1,R,\Delta,\rho,\varepsilon,\delta$.

Our assumptions:
\begin{assumption}[Convexity and initialization]
The function $f$ is Fr\'echet differentiable everywhere on $E$ and satisfies
\[
f((1-t)x+ty)\leq(1-t)f(x)+tf(y),
\qquad x,y\in E,\quad t\in[0,1].
\]
There exists $x^\star\in E$ such that
\[
\begin{aligned}
f(x^\star)&\leq f(x) &&\text{for every }x\in E,\\
\|x^0-x^\star\|&\leq R,\\
f(x^0)-f(x^\star)&\leq\Delta.
\end{aligned}
\]
\end{assumption}

\begin{assumption}[Gap-dependent upper model]
For every $x,y\in E$ satisfying
\[
(2\sqrt{3}+\sqrt{6})\sqrt{H_1}\,\|y-x\|\leq1,
\]
the objective obeys
\[
\begin{aligned}
f(y)\leq{}&f(x)+\langle\nabla f(x),y-x\rangle\\
&+\bigl(H_0+H_1(f(x)-f(x^\star))\bigr)\|y-x\|^2.
\end{aligned}
\]
\end{assumption}

\begin{assumption}[Strong-growth stochastic oracle]
There exist a probability measure $\nu$ on $\Omega$ and a jointly measurable map $g:E\times\Omega\to E$. For every $x\in E$, the map $g(x,\cdot)$ is Bochner integrable and $\|g(x,\cdot)-\nabla f(x)\|^2$ is integrable under $\nu$, with
\[
\begin{aligned}
\mathbb{E}_{\xi\sim\nu}[g(x,\xi)]&=\nabla f(x),\\
\mathbb{E}_{\xi\sim\nu}\bigl[\|g(x,\xi)-\nabla f(x)\|^2\bigr]
&\leq(\rho-1)\|\nabla f(x)\|^2.
\end{aligned}
\]
At call $k\geq0$, a query $x\in E$ receives $g(x,\xi_k)$, where the measurable samples $(\xi_k)_{k\geq0}$ are mutually independent with common law $\nu$. Algorithm randomness is independent of the entire sample sequence.
\end{assumption}

After the allotted oracle calls, the algorithm must return a measurable output $\bar{x}\in E$ with
\[
P\bigl\{f(\bar{x})-f(x^\star)>\varepsilon\bigr\}\leq\delta.
\]

\section*{Convex Minimization with Symmetric Remainders}
We consider minimizing a differentiable convex function $f:\mathbb{R}^d\to\mathbb{R}$, where $d\geq1$, with an $L$-Lipschitz gradient, $L>0$. A given point $x^0$ lies within distance $R>0$ of a global minimizer $x^\star$. The oracle returns $g(x,\xi)$ at $x$, using independent samples $\xi$ from a fixed probability distribution. Each coordinate error has a distribution decomposable into a symmetric measure and a centered remainder with bounded second moment, together with a uniform tail-quantile bound; the symmetric component may be heavy-tailed. For $\varepsilon>0$, the goal is a measurable output $\bar{x}$ with integrable objective gap satisfying $\mathbb{E}[f(\bar{x})-f(x^\star)]\leq\varepsilon$, averaging over oracle and algorithm randomness.

The optimization problem is
\[
\min_{x\in\mathbb{R}^d} f(x),
\]
where $\|\cdot\|$ denotes the Euclidean norm. Fix a measurable sample space $\Omega$ and noise parameters $r>0$ and $M\geq0$. The algorithm receives $d$, $\Omega$, $x^0\in\mathbb{R}^d$, $L$, $R$, $r$, $M$, and $\varepsilon$.

Our assumptions:
\begin{assumption}[Convexity and smoothness]
The function $f$ is differentiable everywhere, and for every $x,y\in\mathbb{R}^d$ and $t\in[0,1]$,
\[
\begin{aligned}
f((1-t)x+ty)&\leq(1-t)f(x)+tf(y),\\
\|\nabla f(x)-\nabla f(y)\|&\leq L\|x-y\|.
\end{aligned}
\]
\end{assumption}

\begin{assumption}[Initial distance]
There exists $x^\star\in\mathbb{R}^d$ such that
\[
\begin{aligned}
f(x^\star)&\leq f(x) &&\text{for every }x\in\mathbb{R}^d,\\
\|x^0-x^\star\|&\leq R.
\end{aligned}
\]
\end{assumption}

\begin{assumption}[Sampling and oracle]
There is a probability measure $\nu$ on $\Omega$ and a jointly measurable map $g:\mathbb{R}^d\times\Omega\to\mathbb{R}^d$. At call $k\geq0$, a query $x\in\mathbb{R}^d$ receives $g(x,\xi_k)$. Under a joint probability law $Q$ on $\Omega^{\mathbb{N}}$, the coordinate samples $\xi_k$ are measurable, mutually independent, and each has distribution $\nu$. Algorithm randomness is independent of this sample sequence.
\end{assumption}

\begin{assumption}[Symmetric remainder and quantile bound]
For every $x\in\mathbb{R}^d$ and $i\in\{1,\ldots,d\}$, let $\mu_{x,i}$ be the distribution of the scalar error $g_i(x,\xi)-[\nabla f(x)]_i$ for $\xi\sim\nu$. There exist finite nonnegative Borel measures $S_{x,i}$ and $A_{x,i}$ on $\mathbb{R}$ such that
\[
\begin{aligned}
\mu_{x,i}&=S_{x,i}+A_{x,i},\\
S_{x,i}(B)&=S_{x,i}(-B)
&&\text{for every Borel }B\subseteq\mathbb{R},
\end{aligned}
\]
where $-B=\{-z:z\in B\}$. The functions $z\mapsto z$ and $z\mapsto z^2$ are integrable with respect to $A_{x,i}$, and
\[
\begin{aligned}
\int_{\mathbb{R}} z\,dA_{x,i}(z)&=0,\\
\int_{\mathbb{R}} z^2\,dA_{x,i}(z)&\leq M,\\
\mu_{x,i}\bigl(\{z\in\mathbb{R}:|z|>r\}\bigr)&\leq\frac{1}{64}.
\end{aligned}
\]
\end{assumption}

After the allotted oracle calls, the output $\bar{x}\in\mathbb{R}^d$ must be measurable, its objective gap must be integrable, and it must satisfy
\[
\mathbb{E}_{P}\bigl[f(\bar{x})-f(x^\star)\bigr]\leq\varepsilon,
\]
where $P$ is the joint law of the algorithm randomness and the oracle sample sequence.

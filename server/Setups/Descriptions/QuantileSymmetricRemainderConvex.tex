\section*{Convex Optimization with Symmetric-Remainder Noise}
We consider minimizing a differentiable convex function $f:\mathbb{R}^d\to\mathbb{R}$, where $d\geq1$, with $L$-Lipschitz gradient and an initial point $x^0$ within Euclidean distance $R$ of a global minimizer $x^\star$. Each oracle call returns a random vector with independent sampling across calls. Each coordinate error has a law decomposing into a symmetric measure, possibly heavy-tailed, and a centered remainder measure with second moment at most $M$; its probability of magnitude exceeding $r$ is at most $1/64$. Here $L,R,r>0$ and $M\geq0$. For $\varepsilon>0$, the goal is a measurable output $\bar{x}$ with integrable objective gap satisfying $\mathbb{E}[f(\bar{x})-f(x^\star)]\leq\varepsilon$, where expectation includes oracle and algorithm randomness.

The optimization problem is
\[
\min_{x\in\mathbb{R}^d} f(x),
\]
with the standard Euclidean inner product and norm. The public inputs are the positive integer $d$, a measurable sample space $\Omega$, the point $x^0\in\mathbb{R}^d$, and the parameters $L,R,r,M,\varepsilon$ with the ranges specified above. The objective and oracle satisfy the following conditions.

\begin{assumption}[Convexity and smoothness]
The function $f$ is differentiable everywhere, and, for every $x,y\in\mathbb{R}^d$ and $t\in[0,1]$,
\[
\begin{aligned}
f((1-t)x+ty) &\leq (1-t)f(x)+tf(y),\\
\|\nabla f(x)-\nabla f(y)\| &\leq L\|x-y\|.
\end{aligned}
\]
\end{assumption}

\begin{assumption}[Attainment and initial distance]
There exists $x^\star\in\mathbb{R}^d$ such that
\[
\begin{aligned}
f(x^\star) &\leq f(x) \qquad \text{for every }x\in\mathbb{R}^d,\\
\|x^0-x^\star\| &\leq R.
\end{aligned}
\]
\end{assumption}

\begin{assumption}[Oracle access and independent sampling]
There are a probability measure $p$ on $\Omega$, a probability measure $P$ on $\Omega^{\mathbb{N}}$, and a jointly measurable map
\[
g:\mathbb{R}^d\times\Omega\to\mathbb{R}^d.
\]
Under $P$, the coordinate samples $\xi_k(\omega)=\omega_k$, for $k\in\mathbb{N}$, are measurable and mutually independent, with
\[
P\circ\xi_k^{-1}=p \qquad \text{for every }k\in\mathbb{N}.
\]
At call $k$, oracle $0$ returns $g(x,\xi_k)$ at the query point $x$. Each vector reply costs one oracle call. Queries may be chosen adaptively anywhere in $\mathbb{R}^d$, and any internal algorithm randomness is independent of the entire sample sequence.
\end{assumption}

\begin{assumption}[Coordinatewise symmetric remainder and quantile bound]
For every $x\in\mathbb{R}^d$ and $i\in\{1,\ldots,d\}$, define the scalar error and its law by
\[
\begin{aligned}
e_{x,i}(\omega) &= \bigl(g(x,\omega)-\nabla f(x)\bigr)_i,\\
\mu_{x,i} &= p\circ e_{x,i}^{-1}.
\end{aligned}
\]
There exist finite nonnegative Borel measures $S_{x,i}$ and $A_{x,i}$ on $\mathbb{R}$ such that, writing $T(z)=-z$,
\[
\begin{aligned}
\mu_{x,i} &= S_{x,i}+A_{x,i},\\
S_{x,i}\circ T^{-1} &= S_{x,i},\\
\int_{\mathbb{R}} |z|\,\mathrm{d}A_{x,i}(z) &<\infty,\\
\int_{\mathbb{R}} z\,\mathrm{d}A_{x,i}(z) &=0,\\
\int_{\mathbb{R}} z^2\,\mathrm{d}A_{x,i}(z) &<\infty,\\
\int_{\mathbb{R}} z^2\,\mathrm{d}A_{x,i}(z) &\leq M.
\end{aligned}
\]
The full coordinate-error law also satisfies
\[
\mu_{x,i}\bigl(\{z\in\mathbb{R}:|z|>r\}\bigr)\leq\frac{1}{64}.
\]
\end{assumption}

The algorithm accesses the objective through oracle $0$ and is selected using only the public inputs $(d,\Omega,x^0,L,R,r,M,\varepsilon)$. A call budget $K\in\mathbb{N}$ may depend on this tuple and counts every oracle reply.

After at most $K$ calls, the algorithm returns a measurable point $\bar{x}\in\mathbb{R}^d$. Let $\mathbb{P}$ denote the joint law of its internal randomness and the oracle samples. Success requires a global minimizer $x^\star$ for which the objective gap is integrable and
\[
\begin{aligned}
\int \bigl|f(\bar{x}(\omega))-f(x^\star)\bigr|\,\mathrm{d}\mathbb{P}(\omega) &<\infty,\\
\int \bigl(f(\bar{x}(\omega))-f(x^\star)\bigr)\,\mathrm{d}\mathbb{P}(\omega) &\leq\varepsilon.
\end{aligned}
\]
The minimizer witnessing the initial distance bound may be used here, since every global minimizer has the same objective value.

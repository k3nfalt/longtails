\section*{Adam with Heavy-Tailed Gradient Noise}
We consider nonconvex optimization of a differentiable function $f:\mathbb{R}^d\to\mathbb{R}$, where $d\geq0$ is an integer and the norm is Euclidean. Its gradient is $L$-Lipschitz, with $L>0$, and a given point $x^0$ satisfies $f(x^0)-f(x)\leq\Delta$ for every $x$, with $\Delta\geq0$. Each oracle call returns an independent-sample unbiased gradient estimate whose noise has summed coordinatewise $\alpha$-moments bounded by $\sigma^\alpha$, where $1<\alpha\leq2$ and $\sigma\geq0$. The algorithm must use the Adam variant specified below and return a measurable random point $\bar{x}$ satisfying $\mathbb{E}\|\nabla f(\bar{x})\|\leq\varepsilon$, where $\varepsilon>0$ and expectation includes both oracle and algorithm randomness.

The underlying optimization problem is
\[
\inf_{x\in\mathbb{R}^d} f(x),
\]
with accuracy measured by the expected gradient norm. The public inputs are $d$, a measurable sample space $\Omega$, the parameters $L,\sigma,\alpha,\Delta,\varepsilon$, and $x^0\in\mathbb{R}^d$, with ranges as stated above. The objective and the oracle law are accessed only through gradient queries. The following conditions specify the admissible objectives and oracles.

\begin{assumption}[Differentiability and smoothness]
The function $f$ is differentiable everywhere, and its Euclidean gradient satisfies
\[
\|\nabla f(x)-\nabla f(y)\|\leq L\|x-y\|
\qquad\text{for all }x,y\in\mathbb{R}^d.
\]
\end{assumption}

\begin{assumption}[Initial objective gap]
The initial point satisfies the global lower-bound condition
\[
f(x^0)-f(x)\leq\Delta
\qquad\text{for all }x\in\mathbb{R}^d.
\]
Thus $f(x^0)-\Delta$ is a lower bound on the objective, without requiring attainment.
\end{assumption}

\begin{assumption}[Independent sampling and oracle access]
There exist a probability measure $p$ on $\Omega$, a probability measure $P$ on $\Omega^{\mathbb{N}}$, and a jointly measurable map
\[
g:\mathbb{R}^d\times\Omega\longrightarrow\mathbb{R}^d.
\]
Under $P$, the coordinate samples $\xi_k$, $k\geq0$, are measurable and mutually independent, with common law $p$. At call $k$, oracle $0$ returns $g(x,\xi_k)$ at the queried point $x\in\mathbb{R}^d$. Each vector reply costs one oracle call. Queries may depend on previous replies, and algorithm randomness is independent of the entire sample sequence.
\end{assumption}

\begin{assumption}[Unbiasedness and heavy-tailed moments]
For every fixed $x\in\mathbb{R}^d$, the map $g(x,\cdot)$ is Bochner integrable under $p$, and
\[
\int_{\Omega}g(x,\omega)\,p(d\omega)=\nabla f(x).
\]
The scalar map $\omega\mapsto\sum_{i=1}^d|g_i(x,\omega)-\nabla_i f(x)|^\alpha$ is also integrable under $p$, and
\[
\int_{\Omega}\sum_{i=1}^d
|g_i(x,\omega)-\nabla_i f(x)|^\alpha\,p(d\omega)
\leq\sigma^\alpha.
\]
\end{assumption}

The permitted algorithms choose, using only the public inputs, constants
\[
0\leq\beta<1,\qquad 0<\rho\leq1,\qquad
\eta>0,\qquad\tau>0,\qquad n\in\mathbb{N}_0.
\]
Starting from $m^0=v^0=0$ and the given $x^0$, each call uses $G^k=g(x^k,\xi_k)$ and performs the coordinatewise updates
\[
\begin{aligned}
m^{k+1}&=\beta m^k+(1-\beta)G^k,\\
v_i^{k+1}&=(1-\rho)v_i^k+\rho(G_i^k)^2,\\
x_i^{k+1}&=x_i^k-\eta\frac{m_i^{k+1}}{\sqrt{v_i^{k+1}}+\tau},
\qquad i=1,\ldots,d.
\end{aligned}
\]
These recursions specify the permitted Adam variant exactly.

Independently of the samples, the algorithm draws $J$ uniformly from $\{0,\ldots,n\}$. For a call budget $K\in\mathbb{N}_0$, it performs $K$ updates and returns
\[
\bar{x}=\begin{cases}
x^J,&J<K,\\
x^0,&J\geq K.
\end{cases}
\]
In particular, the prescribed full run uses $K=n+1$ oracle calls and returns the uniformly selected point among $x^0,\ldots,x^n$; the call at $x^n$ is included in this cost. The budget may depend on the public parameter tuple.

Let $\mathbb{P}$ denote the joint law of $J$ and the oracle sample sequence, namely the product of the uniform law and $P$. Success requires that $\bar{x}$ be measurable, that $\|\nabla f(\bar{x})\|$ be integrable under $\mathbb{P}$, and that
\[
\int\|\nabla f(\bar{x})\|\,d\mathbb{P}\leq\varepsilon.
\]

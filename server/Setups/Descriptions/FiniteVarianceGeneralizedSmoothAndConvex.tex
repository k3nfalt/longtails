\section*{Finite-Variance Generalized Smooth Convex Minimization}
We consider minimizing a convex function $f:E\to\mathbb{R}$ on a separable real Hilbert space $E$. The function $f$ is $(L_0,L_1)$-smooth, with $L_0>0$ and $L_1\geq0$. A given point $x^0\in E$ has distance at most $R>0$ and objective gap at most $\Delta>0$ from a global minimizer $x^\star$. The oracle returns an unbiased stochastic gradient with variance at most $\sigma^2$, where $\sigma\geq0$, using independent, identically distributed samples. For $\varepsilon>0$ and $\delta\in(0,1)$, the goal is a measurable output $\bar{x}\in E$ satisfying $P(f(\bar{x})-f(x^\star)>\varepsilon)\leq\delta$, where $P$ is the joint law of the samples and independent algorithm randomness.

The optimization problem is
\[
\min_{x\in E} f(x).
\]
The norm $\|\cdot\|$ is induced by the inner product on $E$, and $\|\cdot\|_{\mathrm{op}}$ denotes the induced operator norm. Let $\Omega$ be the measurable sample space. The algorithm receives $E$, $\Omega$, $x^0$, $L_0$, $L_1$, $R$, $\Delta$, $\sigma$, $\varepsilon$, and $\delta$; the objective and sampling distribution are accessed through the oracle.

Our assumptions:
\begin{assumption}[Convexity and generalized smoothness]
The function $f$ is Fr\'echet differentiable on $E$, and its gradient $\nabla f:E\to E$ is Fr\'echet differentiable. For every $x,y\in E$ and $t\in[0,1]$,
\[
f((1-t)x+ty)\leq (1-t)f(x)+t f(y).
\]
Generalized smoothness means that
\[
\|\nabla^2 f(x)\|_{\mathrm{op}}
\leq L_0+L_1\|\nabla f(x)\|,
\qquad x\in E,
\]
where $\nabla^2 f(x)$ is a bounded linear operator from $E$ to $E$.
\end{assumption}

\begin{assumption}[Initial distance and objective gap]
There exists $x^\star\in E$ such that
\[
\begin{aligned}
f(x^\star)&\leq f(x) &&\text{for every }x\in E,\\
\|x^0-x^\star\|&\leq R,\\
f(x^0)-f(x^\star)&\leq\Delta.
\end{aligned}
\]
\end{assumption}

\begin{assumption}[Finite-variance stochastic gradients]
There exist a probability measure $\nu$ on $\Omega$ and a jointly measurable map $g:E\times\Omega\to E$. For every $x\in E$, the map $g(x,\cdot)$ is Bochner integrable under $\nu$, the squared error $\|g(x,\cdot)-\nabla f(x)\|^2$ is integrable, and
\[
\begin{aligned}
\mathbb{E}_{\xi\sim\nu}[g(x,\xi)]&=\nabla f(x),\\
\mathbb{E}_{\xi\sim\nu}
\bigl[\|g(x,\xi)-\nabla f(x)\|^2\bigr]&\leq\sigma^2.
\end{aligned}
\]
At call $k\geq0$, a query $x\in E$ receives $g(x,\xi_k)$. The sample variables $(\xi_k)_{k\geq0}$ are measurable and mutually independent, each with law $\nu$, and their joint law is independent of the algorithm's randomness.
\end{assumption}

Queries may be made at any point of $E$. After the allotted oracle calls, the algorithm must return a measurable output $\bar{x}$ satisfying
\[
P\bigl(f(\bar{x})-f(x^\star)>\varepsilon\bigr)\leq\delta.
\]

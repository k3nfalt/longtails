\section*{Quadratic Minimization with Temporal Discrepancy}
We consider minimizing $f(x)=\|A(x-x^\star)\|^2/2$ on a real Hilbert space $E$, where $A:E\to E$ is bounded and linear and $x^\star\in E$. For constants $0<\mu\leq L$, assume $\mu\|v\|^2\leq\|Av\|^2\leq L\|v\|^2$ for every $v\in E$. A given starting point $x^0\in E$ satisfies $f(x^0)\leq\Delta$, where $\Delta\geq0$. At call $k\geq0$, the oracle returns $\nabla f(x)+e_k$, with fixed errors $e_k\in E$ whose partial sums have norm at most $D\geq0$. For $\varepsilon>0$, the goal is an output $\bar{x}\in E$ satisfying $f(\bar{x})-f(x^\star)\leq\varepsilon$ almost surely under $P$, the law of the algorithm's randomness.

The optimization problem is
\[
\min_{x\in E} f(x).
\]
The norm is induced by the inner product on $E$, and $A^*$ denotes the Hilbert-space adjoint of $A$. The algorithm receives only $E$, $L$, $\mu$, and $x^0$; the discrepancy bound $D$, initial objective bound $\Delta$, and tolerance $\varepsilon$ may enter the oracle-call budget but are not algorithm inputs.

Our assumptions:
\begin{assumption}[Quadratic structure]
The objective and its gradient are
\[
\begin{aligned}
f(x)&=\frac12\|A(x-x^\star)\|^2,\\
\nabla f(x)&=A^*A(x-x^\star),
\end{aligned}
\qquad x\in E,
\]
where $A:E\to E$ is a bounded linear operator satisfying
\[
\mu\|v\|^2\leq\|Av\|^2\leq L\|v\|^2,
\qquad v\in E.
\]
In particular, $x^\star$ is the unique minimizer and $f(x^\star)=0$.
\end{assumption}

\begin{assumption}[Initial objective bound]
The given starting point satisfies
\[
f(x^0)\leq\Delta.
\]
\end{assumption}

\begin{assumption}[Temporal discrepancy]
There is a fixed deterministic sequence $(e_k)_{k\geq0}$ in $E$ such that the oracle reply at call $k$, for any query $x\in E$, is
\[
g_k(x)=A^*A(x-x^\star)+e_k.
\]
Its cumulative error satisfies
\[
\left\|\sum_{k=0}^{n-1}e_k\right\|\leq D,
\qquad n=0,1,2,\ldots,
\]
with the empty sum equal to zero.
\end{assumption}

After the allotted oracle calls, the output must satisfy
\[
f(\bar{x})-f(x^\star)\leq\varepsilon
\quad P\text{-a.s.}
\]

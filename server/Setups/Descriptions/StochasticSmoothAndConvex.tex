\section*{Stochastic Smooth Convex Minimization}
We consider minimizing a differentiable convex function $f:E\to\mathbb{R}$ on a separable real Hilbert space $E$. Its gradient is $L$-Lipschitz, where $L>0$, and a given point $x^0\in E$ lies within distance $R\geq0$ of a global minimizer $x^\star$. At $x$, the oracle returns an unbiased stochastic gradient with variance at most $\sigma^2$, where $\sigma\geq0$, using independent samples across calls. For $\varepsilon>0$, the goal is a measurable output $\bar{x}\in E$ with integrable objective gap satisfying $\mathbb{E}[f(\bar{x})-f(x^\star)]\leq\varepsilon$, where expectation includes both algorithm and oracle randomness.

The optimization problem is
\[
\min_{x\in E} f(x).
\]
The norm $\|\cdot\|$ is induced by the inner product on $E$. Let $\Omega$ be the measurable sample space. The algorithm receives $E$, $\Omega$, $L$, $R$, $\sigma$, $\varepsilon$, and $x^0$; the objective and sampling distribution are accessed through the oracle.

Our assumptions:
\begin{assumption}[Convexity]
For every $x,y\in E$ and $t\in[0,1]$,
\[
f((1-t)x+ty)\leq(1-t)f(x)+t f(y).
\]
\end{assumption}

\begin{assumption}[Smoothness]
The function $f$ is Fr\'echet differentiable everywhere on $E$, and its gradient satisfies
\[
\|\nabla f(x)-\nabla f(y)\|\leq L\|x-y\|,
\qquad x,y\in E.
\]
\end{assumption}

\begin{assumption}[Initial distance]
There exists $x^\star\in E$ such that
\[
\begin{aligned}
f(x^\star)&\leq f(x) &&\text{for every }x\in E,\\
\|x^0-x^\star\|&\leq R.
\end{aligned}
\]
\end{assumption}

\begin{assumption}[Stochastic gradients]
There exist a probability measure $p$ on $\Omega$ and a jointly measurable map $g:E\times\Omega\to E$. For every $x\in E$, the map $g(x,\cdot)$ is Bochner integrable under $p$, the squared error $\|g(x,\cdot)-\nabla f(x)\|^2$ is integrable under $p$, and
\[
\begin{aligned}
\mathbb{E}_{\xi\sim p}[g(x,\xi)]&=\nabla f(x),\\
\mathbb{E}_{\xi\sim p}\bigl[\|g(x,\xi)-\nabla f(x)\|^2\bigr]&\leq\sigma^2.
\end{aligned}
\]
At call $k$, a query $x_k\in E$ receives $g(x_k,\xi_k)$, where $(\xi_k)_{k\geq0}$ are independent with common law $p$ and are independent of the algorithm's randomness.
\end{assumption}

Queries may be made at any point of $E$ and may depend on previous replies. After the allotted oracle calls, the output $\bar{x}$ must be measurable and its objective gap must be integrable, with
\[
\mathbb{E}_{P}\bigl[f(\bar{x})-f(x^\star)\bigr]\leq\varepsilon,
\]
where $P$ is the joint law of the algorithm's randomness and the oracle samples.

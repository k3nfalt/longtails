\section*{Convex Minimization with Markov Noise}
We consider minimizing a differentiable convex function $f:\mathbb{R}^d\to\mathbb{R}$, where $d$ is a positive integer, with an $L$-Lipschitz gradient, $L>0$. A given point $x^0$ lies within distance $R>0$ of a global minimizer $x^\star$. At a query $x$, the oracle returns $g(x,Z_k)$, where $k$ indexes calls and $(Z_k)$ is a stationary real-valued Markov chain with stretched-exponential mixing in averaged total variation. The noise is unbiased under the stationary law and has a uniformly bounded moment of order between one and two. For $\varepsilon>0$, the goal is a measurable output $\bar{x}\in\mathbb{R}^d$ with integrable objective gap satisfying $\mathbb{E}[f(\bar{x})-f(x^\star)]\leq\varepsilon$, averaging over the chain and independent algorithm randomness.

The optimization problem is
\[
\min_{x\in\mathbb{R}^d} f(x),
\]
where $\|\cdot\|$ denotes the Euclidean norm. The parameters satisfy
\[
\begin{gathered}
d\in\mathbb{N},\quad d\geq 1,\qquad L,R,\sigma,\varepsilon>0,\\
1<\alpha\leq 2,\qquad \tau\geq 1,\qquad 0<\gamma\leq 1,\\
4\varepsilon\leq LR^2,\qquad 32\varepsilon\leq\sigma R,\\
\frac{LR^2}{\varepsilon}
\leq\left(\frac{\sigma R}{\varepsilon}\right)^{\alpha/(\alpha-1)}.
\end{gathered}
\]
The algorithm receives $d,L,R,\sigma,\alpha,\tau,\gamma,\varepsilon$, and $x^0\in\mathbb{R}^d$.

Our assumptions:
\begin{assumption}[Objective and initial distance]
The function $f$ is differentiable everywhere, and for every $x,y\in\mathbb{R}^d$ and $s\in[0,1]$,
\[
\begin{aligned}
f((1-s)x+sy)&\leq (1-s)f(x)+sf(y),\\
\|\nabla f(x)-\nabla f(y)\|&\leq L\|x-y\|.
\end{aligned}
\]
There exists $x^\star\in\mathbb{R}^d$ such that
\[
\begin{aligned}
f(x^\star)&\leq f(x)\qquad\text{for all }x\in\mathbb{R}^d,\\
\|x^0-x^\star\|&\leq R.
\end{aligned}
\]
\end{assumption}

\begin{assumption}[Stationary Markov sampling and mixing]
Let $\kappa$ be a Markov transition kernel on $\mathbb{R}$ and let $\pi$ be a probability measure on its Borel sets. The samples form a Markov chain with initial law $Z_0\sim\pi$ and transition kernel $\kappa$. The measure $\pi$ is invariant: for every Borel set $A\subseteq\mathbb{R}$,
\[
\mathbb{E}_{Z\sim\pi}[\kappa(Z,A)]=\pi(A).
\]
For each integer $t\geq 0$, define
\[
\begin{aligned}
m_t(z)&=\|\kappa^t(z,\cdot)-\pi\|_{\mathrm{TV}},\\
\|\mu-\nu\|_{\mathrm{TV}}
&=\sup_{A\in\mathcal{B}(\mathbb{R})}|\mu(A)-\nu(A)|,
\end{aligned}
\]
where $\mathcal{B}(\mathbb{R})$ denotes the Borel sets and $\kappa^0(z,\cdot)$ is the point mass at $z$. The function $m_t$ is $\pi$-integrable and satisfies
\[
\mathbb{E}_{Z\sim\pi}[m_t(Z)]
\leq \exp\!\left(-\left(\frac{t}{\tau}\right)^\gamma\right),
\qquad t\geq 0.
\]
\end{assumption}

\begin{assumption}[Stationary oracle moments]
The map $g:\mathbb{R}^d\times\mathbb{R}\to\mathbb{R}^d$ is jointly measurable. For every fixed $x\in\mathbb{R}^d$, the map $g(x,\cdot)$ is Bochner integrable under $\pi$, the function $z\mapsto\|g(x,z)-\nabla f(x)\|^\alpha$ is $\pi$-integrable, and
\[
\begin{aligned}
\mathbb{E}_{Z\sim\pi}[g(x,Z)]&=\nabla f(x),\\
\mathbb{E}_{Z\sim\pi}
\bigl[\|g(x,Z)-\nabla f(x)\|^\alpha\bigr]
&\leq\sigma^\alpha.
\end{aligned}
\]
\end{assumption}

Queries may be made at any point of $\mathbb{R}^d$. On call $k=0,1,\ldots$, a query $x_k$ receives $g(x_k,Z_k)$; successive calls use successive states of the same chain. The algorithm's randomness is independent of the chain. After the allotted calls, its output $\bar{x}$ must be measurable, the gap $f(\bar{x})-f(x^\star)$ must be integrable, and
\[
\mathbb{E}_{P}\bigl[f(\bar{x})-f(x^\star)\bigr]
\leq\varepsilon,
\]
where $P$ is the joint law of the chain and the algorithm's randomness.

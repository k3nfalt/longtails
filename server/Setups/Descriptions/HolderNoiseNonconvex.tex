\section*{Nonconvex Optimization with H\"older Noise}
We consider minimizing a differentiable, possibly nonconvex function $f:E\to\mathbb{R}$ on a separable real Hilbert space $E$. Its gradient is $L$-Lipschitz, with $L>0$, and a given point $x^0\in E$ satisfies $f(x^0)-\inf_{x\in E}f(x)\leq\Delta$, with $\Delta\geq0$. Each oracle call returns unbiased gradients at two queried points using one shared sample, with independent samples across calls. The centered noise has variance at most $\sigma^2$ and mean-square H\"older increments bounded by $M\|x-y\|^{2\nu}$, where $\sigma,M\geq0$ and $0<\nu\leq1$. The goal is a measurable output $\bar{x}\in E$ satisfying $\mathbb{E}\|\nabla f(\bar{x})\|\leq\varepsilon$, where $\varepsilon>0$ and expectation includes sampling and algorithm randomness.

The optimization problem is
\[
\inf_{x\in E} f(x),
\]
where $\|\cdot\|$ is the norm induced by the inner product on $E$. The algorithm receives $E$, the measurable sample space $\Omega$, $L$, $M$, $\nu$, $\sigma$, $\Delta$, $\varepsilon$, and $x^0$.

Our assumptions:
\begin{assumption}[Smoothness and initial gap]
The function $f$ is Fr\'echet differentiable everywhere, and for every $x,y\in E$,
\[
\begin{aligned}
\|\nabla f(x)-\nabla f(y)\|&\leq L\|x-y\|,\\
f(x^0)-f(x)&\leq\Delta.
\end{aligned}
\]
\end{assumption}

\begin{assumption}[Unbiased stochastic gradients]
There exist a probability measure $p$ on $\Omega$ and a jointly measurable map $g:E\times\Omega\to E$. For every $x\in E$, the random vector $g(x,\xi)$ is Bochner integrable and $\|g(x,\xi)-\nabla f(x)\|^2$ is integrable under $\xi\sim p$, with
\[
\begin{aligned}
\mathbb{E}_{\xi\sim p}[g(x,\xi)]&=\nabla f(x),\\
\mathbb{E}_{\xi\sim p}\bigl[\|g(x,\xi)-\nabla f(x)\|^2\bigr]
&\leq\sigma^2.
\end{aligned}
\]
\end{assumption}

\begin{assumption}[H\"older noise increments]
Define the centered noise by $\zeta(x,\xi)=g(x,\xi)-\nabla f(x)$. For every $x,y\in E$, the squared increment $\|\zeta(x,\xi)-\zeta(y,\xi)\|^2$ is integrable under $\xi\sim p$ and satisfies
\[
\mathbb{E}_{\xi\sim p}\bigl[\|\zeta(x,\xi)-\zeta(y,\xi)\|^2\bigr]
\leq M\|x-y\|^{2\nu}.
\]
\end{assumption}

At call $k$, a query $(x_k,y_k)\in E\times E$ receives
\[
\bigl(g(x_k,\xi_k),\,g(y_k,\xi_k)\bigr).
\]
The sample sequence $(\xi_k)_{k\geq0}$ consists of measurable, mutually independent random variables with common law $p$. Algorithm randomness is independent of the entire sample sequence. Queries may depend on earlier replies, and each paired reply counts as one oracle call.

After the allotted calls, the output $\bar{x}$ must be measurable, its gradient norm must be integrable, and
\[
\mathbb{E}_{P}\bigl[\|\nabla f(\bar{x})\|\bigr]\leq\varepsilon,
\]
where $P$ is the joint law of the sample sequence and the algorithm's randomness.

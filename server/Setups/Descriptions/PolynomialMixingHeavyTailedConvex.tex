\section*{Convex Minimization with Polynomial Mixing}
We consider minimizing a differentiable convex function $f:E\to\mathbb{R}$ on a separable real Hilbert space $E$. Its gradient is $L$-Lipschitz, with $L>0$, and a given point $x^0\in E$ lies within distance $R>0$ of a global minimizer $x^\star$. At call $k\geq0$, the oracle returns $g(x,Z_k)\in E$ at query $x\in E$, where $(Z_k)$ is a stationary Markov chain with polynomial mixing. Under its stationary distribution, the replies are unbiased and have uniformly bounded centered moments of order $\alpha\in(1,2]$. For $\varepsilon>0$, the goal is a measurable output $\bar{x}\in E$ with integrable objective gap satisfying $\mathbb{E}[f(\bar{x})-f(x^\star)]\leq\varepsilon$.

The optimization problem is
\[
\min_{x\in E} f(x).
\]
The norm $\|\cdot\|$ is induced by the inner product on $E$. Let $\Omega$ be the measurable state space of the chain. The algorithm receives $E$, $\Omega$, $x^0$, and all numerical parameters $L,R,\sigma,\alpha,\tau,\beta,\varepsilon$; the objective and the oracle's defining map and transition law are accessed only through oracle replies.

Our assumptions:
\begin{assumption}[Parameter regime]
The parameters satisfy
\[
L,R,\sigma,\tau,\beta,\varepsilon>0,
\qquad 1<\alpha\leq2.
\]
There exists a unit vector in $E$, and $\Omega$ contains distinct points $a,b$ whose singletons are measurable. The numerical restrictions are
\[
\begin{aligned}
16&\leq\frac{LR^2}{\varepsilon},\\
\frac{LR^2}{\varepsilon}
&\leq\left(\frac{\sigma R}{\varepsilon}\right)^{\alpha/(\alpha-1)},\\
2^{4\alpha-1}
&\leq\left(\frac{\sigma R}{\varepsilon}\right)^\alpha,\\
\beta\left(
\frac{2^{4\alpha-1}}{(\sigma R/\varepsilon)^\alpha}
\right)^{1/(\beta(\alpha-1))}
&\leq\tau.
\end{aligned}
\]
\end{assumption}

\begin{assumption}[Objective and initial distance]
The function $f$ is convex and Fr\'echet differentiable everywhere on $E$. For every $x,y\in E$ and $s\in[0,1]$,
\[
\begin{aligned}
f((1-s)x+sy)&\leq(1-s)f(x)+s f(y),\\
\|\nabla f(x)-\nabla f(y)\|&\leq L\|x-y\|.
\end{aligned}
\]
There exists $x^\star\in E$ such that
\[
\begin{aligned}
f(x^\star)&\leq f(x)\qquad\text{for every }x\in E,\\
\|x^0-x^\star\|&\leq R.
\end{aligned}
\]
\end{assumption}

\begin{assumption}[Stationary polynomial mixing]
The chain has a Markov transition kernel $\kappa$ on $\Omega$ and initial distribution $\pi$, where $\pi$ is an invariant probability measure. Thus, for every measurable $A\subseteq\Omega$,
\[
Z_0\sim\pi,
\qquad
\mathbb{E}_{Z\sim\pi}[\kappa(Z,A)]=\pi(A).
\]
For every integer $t\geq0$, define
\[
d_t(z)=\sup_{A\subseteq\Omega\text{ measurable}}
\bigl|\kappa^t(z,A)-\pi(A)\bigr|,
\]
where $\kappa^t$ is the $t$-step transition kernel and $\kappa^0(z,\cdot)$ is the probability measure concentrated at $z$. This is the total-variation convention without an additional factor of two. Each $d_t$ is integrable under $\pi$, and
\[
\mathbb{E}_{Z\sim\pi}[d_t(Z)]
\leq\left(1+\frac{t}{\tau}\right)^{-\beta},
\qquad t\geq0.
\]
\end{assumption}

\begin{assumption}[Stationary oracle moments]
The map $g:E\times\Omega\to E$ is jointly measurable. For every fixed $x\in E$, the map $g(x,\cdot)$ is Bochner integrable under $\pi$, the function $\|g(x,\cdot)-\nabla f(x)\|^\alpha$ is integrable under $\pi$, and
\[
\begin{aligned}
\mathbb{E}_{Z\sim\pi}[g(x,Z)]&=\nabla f(x),\\
\mathbb{E}_{Z\sim\pi}
\bigl[\|g(x,Z)-\nabla f(x)\|^\alpha\bigr]
&\leq\sigma^\alpha.
\end{aligned}
\]
\end{assumption}

Queries may be made at any point of $E$. The reply to the $k$th query $x_k$ is $g(x_k,Z_k)$, using consecutive states of the same chain, including when queries are adaptive. Any algorithm randomness is independent of the entire chain. All calls count toward the oracle-call budget, including calls whose replies are discarded.

After the allotted calls, the output $\bar{x}$ must be measurable, its objective gap must be integrable, and it must satisfy
\[
\mathbb{E}_{P}\bigl[f(\bar{x})-f(x^\star)\bigr]
\leq\varepsilon,
\]
where $P$ is the joint law of the algorithm's randomness and the stationary Markov chain.

\section*{Composite Minimization with Lipschitz Hessian}
We consider minimizing $F=f+\psi$ over a convex set $D$ in a finite-dimensional real inner product space $E$. The function $f:E\to\mathbb{R}$ is convex and twice continuously differentiable, with $H$-Lipschitz Hessian, where $H>0$; $\psi:E\to\mathbb{R}$ is convex on $D$ and has a closed epigraph restricted to $D$. A given point $x^0\in D$ satisfies $\|x^0-x^\star\|\leq R$ for a minimizer $x^\star\in D$ and $R>0$. Three oracles return exact gradients, exact Hessians, and exact composite model solutions. For $\varepsilon>0$, the goal is $\bar{x}\in D$ with $F(\bar{x})-F(x^\star)\leq\varepsilon$ almost surely under $P$, the joint law of algorithm and oracle randomness.

Write $F(x)=f(x)+\psi(x)$. The optimization problem is
\[
\min_{x\in D} F(x).
\]
The norm $\|\cdot\|$ is induced by the inner product $\langle\cdot,\cdot\rangle$ on $E$, and $\|\cdot\|_{\mathrm{op}}$ denotes the induced operator norm. The algorithm receives $E$, $x^0$, $H>0$, $R>0$, $\varepsilon>0$, and an integer $m\geq 1$.

Our assumptions:
\begin{assumption}[Convex composite objective]
The function $f$ is convex on $E$, the set $D\subseteq E$ is convex, and $\psi$ is convex on $D$. In particular, for $t\in[0,1]$,
\[
\begin{aligned}
f((1-t)x+ty)&\leq (1-t)f(x)+tf(y),
&&x,y\in E,\\
\psi((1-t)x+ty)&\leq (1-t)\psi(x)+t\psi(y),
&&x,y\in D.
\end{aligned}
\]
The restricted epigraph
\[
\{(x,r)\in E\times\mathbb{R}:x\in D,\ \psi(x)\leq r\}
\]
is closed in $E\times\mathbb{R}$.
\end{assumption}

\begin{assumption}[Lipschitz Hessian]
The function $f$ is twice continuously Fr\'echet differentiable on $E$. Its Hessian, viewed as a linear operator from $E$ to $E$, satisfies
\[
\|\nabla^2 f(x)-\nabla^2 f(y)\|_{\mathrm{op}}
\leq H\|x-y\|,
\qquad x,y\in E.
\]
\end{assumption}

\begin{assumption}[Minimizer and initial distance]
The given starting point $x^0$ and a minimizer $x^\star$ belong to $D$, with
\[
\begin{aligned}
F(x^\star)&\leq F(x) &&\text{for every }x\in D,\\
\|x^0-x^\star\|&\leq R.
\end{aligned}
\]
\end{assumption}

Gradient and Hessian queries may be made at any $x\in E$. The gradient oracle returns $\nabla f(x)$ at $x$, and the Hessian oracle returns $\nabla^2 f(x)$ at $x$.

The third oracle solves composite models. A model query consists of a center $c\in E$, a vector $a\in E$, a linear operator $B:E\to E$, a cubic coefficient $\gamma\geq 0$, and a scale $s>0$. The operator must satisfy
\[
\begin{aligned}
\langle u,Bv\rangle&=\langle Bu,v\rangle
&&\text{for all }u,v\in E,\\
\langle u,Bu\rangle&\geq 0
&&\text{for all }u\in E.
\end{aligned}
\]
In addition, either $\gamma>0$, or there exists $\mu>0$ such that
\[
\langle u,Bu\rangle\geq\mu\|u\|^2,
\qquad u\in E.
\]
For every such query, the model oracle returns a point $z\in D$ satisfying the exact variational inequality
\[
\begin{aligned}
d(z)&=a+B(z-c)+\gamma\|z-c\|(z-c),\\
0&\leq\langle d(z),u-z\rangle
+s\bigl(\psi(u)-\psi(z)\bigr),
\qquad u\in D.
\end{aligned}
\]
Equivalently, $z$ minimizes over $D$ the model
\[
M(x)=\langle a,x-c\rangle
+\frac{1}{2}\langle x-c,B(x-c)\rangle
+\frac{\gamma}{3}\|x-c\|^3+s\psi(x).
\]
The model guarantee holds for every valid query and every oracle reply, including queries chosen using previous replies. Gradient, Hessian, and model-solution calls are counted separately.

After the allotted calls, the output must satisfy
\[
\bar{x}\in D,
\qquad F(\bar{x})-F(x^\star)\leq\varepsilon
\quad P\text{-a.s.}
\]
Under the joint law $P$, the algorithm's randomness is independent of the oracle's randomness.

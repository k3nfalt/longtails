\section*{Near-Zero-Sum Games with Exact Oracles}
We consider a two-player game on nonempty compact convex sets $X\subset E$ and $Y\subset F$, where $E,F$ are finite-dimensional real inner-product spaces. For functions $h,g:E\times F\to\mathbb{R}$, the players minimize $h+g$ in $x\in X$ and $g-h$ in $y\in Y$, respectively. The function $h$ is strongly convex in $x$ and strongly concave in $y$; $g$ is jointly convex, with Lipschitz gradients specified on the feasible set. An exact oracle returns either the players' own-gradient vector or a Euclidean projection. Given $\varepsilon>0$, the output $\bar z$ must be feasible and satisfy $\|\bar z-z^\star\|^2\leq\varepsilon$ almost surely for one fixed equilibrium $z^\star$, using the Euclidean product norm.

Write $Z=E\times F$ and $C=X\times Y$, with
\[
\langle (x,y),(u,v)\rangle
=\langle x,u\rangle_E+\langle y,v\rangle_F,
\qquad
\|(x,y)\|^2=\|x\|_E^2+\|y\|_F^2.
\]
The game seeks a pair $(x^\star,y^\star)\in C$ satisfying
\[
\begin{aligned}
x^\star&\in\operatorname*{arg\,min}_{x\in X}
       \bigl(h(x,y^\star)+g(x,y^\star)\bigr),\\
y^\star&\in\operatorname*{arg\,min}_{y\in Y}
       \bigl(g(x^\star,y)-h(x^\star,y)\bigr).
\end{aligned}
\]
The public data are $E,F,X,Y,L,\mu,\nu,\delta,D,\varepsilon$ and the initial point $z^0=(x^0,y^0)$. Their ranges are
\[
0<\mu\leq L,\qquad 0<\nu\leq L,\qquad
0\leq\delta\leq L,\qquad D>0,\qquad\varepsilon>0.
\]
The functions $h,g$ are instance data accessed through the oracle. The admissible instances satisfy the following conditions.

\begin{assumption}[Feasible geometry]
The sets $X$ and $Y$ are compact and convex, $x^0\in X$, and $y^0\in Y$. Their product has diameter at most $D$:
\[
\|z-w\|\leq D\qquad\text{for all }z,w\in C.
\]
\end{assumption}

\begin{assumption}[Game curvature]
For every $t\in[0,1]$, $x,u\in X$, and $y,v\in Y$, the function $h$ satisfies
\[
\begin{aligned}
h((1-t)x+tu,y)
&\leq (1-t)h(x,y)+t h(u,y)\\
&\quad-\frac{\mu}{2}t(1-t)\|x-u\|_E^2,\\
h(x,(1-t)y+tv)
&\geq (1-t)h(x,y)+t h(x,v)\\
&\quad+\frac{\nu}{2}t(1-t)\|y-v\|_F^2.
\end{aligned}
\]
The function $g$ is convex on $C$: for $z,w\in C$ and $t\in[0,1]$,
\[
g((1-t)z+tw)\leq(1-t)g(z)+t g(w).
\]
\end{assumption}

\begin{assumption}[Equilibrium existence]
There exists $z^\star=(x^\star,y^\star)\in C$ such that, for every $x\in X$ and $y\in Y$,
\[
\begin{aligned}
h(z^\star)+g(z^\star)
&\leq h(x,y^\star)+g(x,y^\star),\\
g(z^\star)-h(z^\star)
&\leq g(x^\star,y)-h(x^\star,y).
\end{aligned}
\]
Any point satisfying these conditions is called an equilibrium.
\end{assumption}

\begin{assumption}[Feasible gradients and smoothness]
There exist maps $H,G:Z\to Z$ that are gradients of $h,g$ within $C$. Explicitly, at every $z\in C$, as $w\to z$ with $w\in C$,
\[
\begin{aligned}
h(w)&=h(z)+\langle H(z),w-z\rangle+o(\|w-z\|),\\
g(w)&=g(z)+\langle G(z),w-z\rangle+o(\|w-z\|).
\end{aligned}
\]
For all $z,w\in C$, these maps satisfy
\[
\begin{aligned}
\|(H+G)(z)-(H+G)(w)\|&\leq L\|z-w\|,\\
\|(H-G)(z)-(H-G)(w)\|&\leq L\|z-w\|,\\
\|G(z)-G(w)\|&\leq\delta\|z-w\|.
\end{aligned}
\]
\end{assumption}

\begin{assumption}[Exact gradient and projection access]
A single oracle, with zero-based oracle ID $0$, accepts two query types. A gradient query at $z\in C$ returns the vector
\[
V(z)=\bigl(H_x(z)+G_x(z),\,-H_y(z)+G_y(z)\bigr)\in Z,
\]
where subscripts denote the $E$ and $F$ components. Thus the second component has the negative sign on the gradient of $h$ and the positive sign on the gradient of $g$.

A projection query at any $z\in Z$ returns $P(z)\in C$, where
\[
\langle z-P(z),w-P(z)\rangle\leq0
\qquad\text{for every }w\in C.
\]
Both replies are exact at every call index and for every realization of oracle randomness. Each query costs one call to oracle $0$; gradient and projection queries share the same call budget. Projection queries may lie outside $C$, while gradient replies are prescribed only on $C$. Any internal algorithm randomness is independent of oracle randomness.
\end{assumption}

Queries may be adaptive. The algorithm is selected using the full public data above, and an externally specified budget $K\in\mathbb{N}$ limits the total number of calls to oracle $0$. Its output is a point $\bar z\in Z$. Writing $\mathbb{P}$ for the joint law of algorithm and oracle randomness, success requires the existence of one equilibrium $z^\star$ such that
\[
\bar z\in C
\quad\text{and}\quad
\|\bar z-z^\star\|^2\leq\varepsilon
\qquad\mathbb{P}\text{-almost surely}.
\]
The equilibrium in this requirement is fixed independently of the realization of randomness.

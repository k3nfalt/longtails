\section*{Spectrally Smooth Stochastic Matrix Optimization}
We consider a differentiable, possibly nonconvex objective $f:E\to\mathbb{R}$ on the Frobenius space $E=\mathbb{R}^{m\times n}$, where $m,n$ are nonnegative integers. Its gradient is $L$-Lipschitz from the spectral norm to the nuclear norm, and an initial point $x^0\in E$ has objective gap at most $\Delta$ above a global lower bound. Here $L,\Delta\geq0$. Each oracle call returns an unbiased stochastic gradient using an independent sample, with rowwise standard deviations summing to at most $s\geq0$ in a fixed hidden orthonormal basis. Given $\varepsilon>0$ and $\delta\in(0,1)$, an admissible momentum-polar algorithm must return a measurable $\bar{x}\in E$ whose gradient has nuclear norm at most $\varepsilon$ with probability at least $1-\delta$.

The underlying optimization problem is
\[
\inf_{x\in E} f(x).
\]
Equip $E$ with the Frobenius inner product $\langle X,Y\rangle_F=\sum_{i,j}X_{ij}Y_{ij}$ and its induced norm $\|\cdot\|_F$. Write $\|\cdot\|_{\mathrm{op}}$ for the operator norm of a matrix acting from Euclidean $\mathbb{R}^n$ to Euclidean $\mathbb{R}^m$, and $\|\cdot\|_*$ for its dual nuclear norm. Gradients are taken with respect to the Frobenius inner product. The public inputs are exactly $(m,n,L,\Delta,s,\varepsilon,\delta,x^0)$, with the ranges stated above.

\begin{assumption}[Spectral smoothness]
The function $f$ is differentiable everywhere, and
\[
\|\nabla f(x)-\nabla f(y)\|_*
\leq L\|x-y\|_{\mathrm{op}}
\qquad (x,y\in E).
\]
\end{assumption}

\begin{assumption}[Lower bound and initialization]
There exists $f_{\mathrm{low}}\in\mathbb{R}$ such that
\[
\begin{aligned}
f_{\mathrm{low}}&\leq f(x) &&(x\in E),\\
f(x^0)-f_{\mathrm{low}}&\leq\Delta.
\end{aligned}
\]
The lower bound need not be attained.
\end{assumption}

\begin{assumption}[Unbiased gradients and rowwise variance]
There exist a measurable sample space $\Omega$, a probability measure $\mu$ on $\Omega$, and a jointly measurable map $g:E\times\Omega\to E$. For every $x\in E$, the map $g(x,\cdot)$ is Bochner integrable and satisfies
\[
\int_{\Omega}g(x,\omega)\,d\mu(\omega)=\nabla f(x).
\]
There also exist a fixed orthonormal basis $(u_i)_{i=1}^m$ of $\mathbb{R}^m$ and numbers $\sigma_i\geq0$ satisfying $\sum_{i=1}^m\sigma_i\leq s$. With $Z(x,\omega)=g(x,\omega)-\nabla f(x)$, for every $x\in E$ and $i=1,\ldots,m$,
\[
\begin{aligned}
Z(x,\cdot)^{\mathsf T}u_i&\in L^2(\mu;\mathbb{R}^n),\\
\int_{\Omega}\|Z(x,\omega)^{\mathsf T}u_i\|_2^2\,d\mu(\omega)
&\leq\sigma_i^2.
\end{aligned}
\]
The basis, the individual variance bounds, the sampling model, and $f_{\mathrm{low}}$ are not public inputs.
\end{assumption}

\begin{assumption}[Independent oracle calls]
Under a probability law $P$ on $\Omega^{\mathbb{N}}$, the coordinate samples $(\xi_t)_{t\geq0}$ are measurable, mutually independent, and each has law $\mu$. At call $t$, the sole oracle, with ID $0$, returns $g(x,\xi_t)\in E$ at the queried point $x\in E$. Each reply costs one oracle call. Internal algorithm randomness is independent of the entire sample sequence. Queries are unrestricted in location within $E$, subject to the algorithm family below.
\end{assumption}

\begin{assumption}[Admissible momentum-polar family]
An admissible algorithm either always returns $x^0$, a choice permitted when $2L\Delta\leq\varepsilon^2$, or follows the procedure specified here. Its parameters obey
\[
0<\alpha\leq1,\qquad \eta>0,\qquad \rho>0,\qquad
T,k,q\in\mathbb{Z}_{\geq0},\qquad b\in\mathbb{Z}_{\geq1}.
\]
For $M\in E$, define $\operatorname{polar}(M)$ as the minimum-Frobenius-norm element of
\[
\underset{\|U\|_{\mathrm{op}}\leq1}{\operatorname{arg\,max}}
\langle M,U\rangle_F.
\]
Starting from $M^0=0$, the training points obey
\[
\begin{aligned}
G^t&=g(x^t,\xi_t),\\
M^{t+1}&=(1-\alpha)M^t+\alpha G^t,\\
x^{t+1}&=x^t-\eta\operatorname{polar}(M^{t+1})
\qquad (0\leq t<T).
\end{aligned}
\]
The algorithm queries all $T+1$ points $x^0,\ldots,x^T$ in this order, including the call at $x^T$.

Using its internal randomness, it draws $q+1$ independent uniform indices $J_0,\ldots,J_q$ from $\{0,\ldots,T\}$, independently of the oracle randomness, and sets $y_i=x^{J_i}$. At each $y_i$, it then obtains $k+1$ batches of $b$ new oracle replies. In increasing order of $i$, then $j$, then $u$, define
\[
\begin{aligned}
a_{ij}&=T+1+b(k+1)i+bj,\\
B_{ij}&=\frac1b\sum_{u=0}^{b-1}
 g(y_i,\xi_{a_{ij}+u}),
\end{aligned}
\]
for $0\leq i\leq q$ and $0\leq j\leq k$. For each candidate, form
\[
S_i=\left\{j\in\{0,\ldots,k\}:
2\left|\left\{\ell\in\{0,\ldots,k\}:
\|B_{ij}-B_{i\ell}\|_*\leq\rho\right\}\right|>k+1\right\}.
\]
Let $r_i$ be the smallest index in $S_i$ when $S_i$ is nonempty, and let $r_i=0$ otherwise. Set $\widehat G_i=B_{i r_i}$ and choose
\[
i_* = \min\underset{0\leq i\leq q}{\operatorname{arg\,min}}
\|\widehat G_i\|_*.
\]
After these calls, the output is $\bar{x}=y_{i_*}$. This output remains the same after any additional replies. The prescribed query and output rules hold for every realization of the algorithm randomness and every possible reply sequence.
\end{assumption}

The algorithm and its tuning choices use only the public inputs. Completing the nonconstant procedure requires
\[
N=T+1+b(k+1)(q+1)
\]
separately counted calls to oracle $0$, including the call at $x^T$ and every batch sample. Any additional calls also count toward the externally supplied budget $K\in\mathbb{Z}_{\geq0}$; the family specifies the selected-candidate output once $N$ calls have been completed.

For the output $\bar{x}$ at the supplied budget, let $\mathbb{P}$ denote the joint law of the independent algorithm and oracle randomness. Success requires that $\bar{x}$ be measurable and that
\[
\mathbb{P}\!\left(\|\nabla f(\bar{x})\|_*>\varepsilon\right)\leq\delta.
\]

\section*{Optimization with Heterogeneous Symmetric Noise}
We consider a differentiable, possibly nonconvex objective $f:E\to\mathbb{R}$ on the Euclidean space $E=\mathbb{R}^{b}\times\mathbb{R}^{h}$, where $b,h\geq0$ are integers with $b+h>0$. Its gradient is $L$-Lipschitz, with $L>0$, and $f(x^0)-\inf_E f\leq\Delta$ for an initial point $x^0\in E$ and $\Delta>0$. Each oracle call returns an unbiased stochastic gradient using an independent sample. Body noise has symmetric coordinate marginals, bounded first moments, and a random Lipschitz modulus with finite mean; head noise has bounded second moments. An admissible heterogeneous Adam method must return a measurable point $\bar{x}$ satisfying $\mathbb{E}\|\nabla f(\bar{x})\|\leq\varepsilon$, where $\varepsilon>0$ and expectation includes all randomness.

The underlying optimization problem is
\[
\inf_{x\in E} f(x).
\]
Write $x=(x_B,x_H)$ for the body and head blocks, with norm
\[
\|x\|^2=\sum_{i=1}^{b}x_{B,i}^2+\sum_{j=1}^{h}x_{H,j}^2.
\]
Either block may be empty, in which case its sums and coordinate conditions are vacuous. The public inputs are $b,h$, a measurable sample space $\Omega$, the initial point $x^0$, and the parameters
\[
L>0,\qquad M,\sigma_B,\sigma_H\geq0,\qquad
\Delta,\varepsilon,\delta>0.
\]
Here $\delta$ is the body denominator offset. The objective and oracle satisfy the following assumptions.

\begin{assumption}[Smoothness and initial gap]
The function $f$ is differentiable everywhere, and, for all $x,y\in E$,
\[
\begin{aligned}
\|\nabla f(x)-\nabla f(y)\|&\leq L\|x-y\|,\\
f(x^0)-f(x)&\leq\Delta.
\end{aligned}
\]
\end{assumption}

\begin{assumption}[Independent unbiased oracle]
There exist a probability measure $p$ on $\Omega$, a probability measure $P$ on $\Omega^{\mathbb{N}}$, and a jointly measurable map $g:E\times\Omega\to E$. Under $P$, the coordinate samples $\xi_t$, $t\geq0$, are measurable, mutually independent, and each has law $p$. For every $x\in E$, the map $g(x,\cdot)$ is Bochner integrable and
\[
\int_{\Omega}g(x,\omega)\,dp(\omega)=\nabla f(x).
\]
Oracle ID $0$ returns $g(x,\xi_t)$ on call $t$ at query $x$, at a cost of one oracle call for the entire vector. Queries may be adaptive and may lie anywhere in $E$. Algorithm randomness is independent of the sample sequence.
\end{assumption}

\begin{assumption}[Heterogeneous moments and symmetry]
Define the noise $z(x,\omega)=g(x,\omega)-\nabla f(x)$, with blocks $z_B$ and $z_H$. For every $x\in E$,
\[
\begin{aligned}
\int_{\Omega}\sum_{i=1}^{b}|z_{B,i}(x,\omega)|\,dp(\omega)
&\leq\sigma_B,\\
\int_{\Omega}\sum_{j=1}^{h}z_{H,j}(x,\omega)^2\,dp(\omega)
&\leq\sigma_H^2.
\end{aligned}
\]
The head squared norm $\omega\mapsto\sum_{j=1}^{h}z_{H,j}(x,\omega)^2$ is integrable. Each body coordinate has a symmetric marginal distribution: for every $i=1,\ldots,b$ and every Borel set $A\subseteq\mathbb{R}$,
\[
p\{\omega:z_{B,i}(x,\omega)\in A\}
=p\{\omega:-z_{B,i}(x,\omega)\in A\}.
\]
\end{assumption}

\begin{assumption}[Body noise regularity]
There exists a measurable, integrable function $K:\Omega\to\mathbb{R}$ such that
\[
\int_{\Omega}K(\omega)\,dp(\omega)\leq M.
\]
On a single set of full $p$-measure, $K(\omega)\geq0$ and, simultaneously for all $x,y\in E$ and $i=1,\ldots,b$,
\[
|z_{B,i}(x,\omega)-z_{B,i}(y,\omega)|
\leq K(\omega)\|x-y\|.
\]
\end{assumption}

The objective is accessed through this oracle. The algorithm may use all public inputs, whereas $f,g,p,P$, and $K$ are not supplied as public inputs. Admissible algorithms are restricted to the following heterogeneous Adam family. Choose a terminal selection index $T\in\mathbb{N}_0$, an initial head scale $r_0>0$, and deterministic positive step sequences $\eta_{B,t},\eta_{H,t}>0$ for every $t\geq0$, using the public inputs. Fix
\[
\alpha=\frac{9}{10},\qquad \beta=\frac{999}{1000}.
\]
Initialize the body momentum and accumulators by
\[
m_{B,i}^{0}=0,\qquad v_{B,i}^{0}=0,\qquad v_{H,j}^{0}=r_0^2.
\]
At iteration $t$, one oracle call gives $G^t=g(x^t,\xi_t)$. For each body coordinate, the update is
\[
\begin{aligned}
m_{B,i}^{t+1}&=\alpha m_{B,i}^{t}+(1-\alpha)G_{B,i}^{t},\\
v_{B,i}^{t+1}&=\beta v_{B,i}^{t}+(1-\beta)(G_{B,i}^{t})^2,\\
\widehat m_{B,i}^{t+1}&=\frac{m_{B,i}^{t+1}}{1-\alpha^{t+1}},\qquad
\widehat v_{B,i}^{t+1}=\frac{v_{B,i}^{t+1}}{1-\beta^{t+1}},\\
x_{B,i}^{t+1}&=x_{B,i}^{t}-\eta_{B,t}
\frac{\widehat m_{B,i}^{t+1}}{\delta+\sqrt{\widehat v_{B,i}^{t+1}}}.
\end{aligned}
\]
For each head coordinate, the update is
\[
\begin{aligned}
v_{H,j}^{t+1}&=v_{H,j}^{t}+(G_{H,j}^{t})^2,\\
x_{H,j}^{t+1}&=x_{H,j}^{t}-\eta_{H,t}
\frac{G_{H,j}^{t}}{\sqrt{v_{H,j}^{t+1}}}.
\end{aligned}
\]
The algorithm draws $J$ uniformly from $\{0,\ldots,T\}$, independently of the oracle randomness. With a call budget $N\in\mathbb{N}_0$, it performs $N$ updates and returns
\[
\bar{x}=\begin{cases}
x^J,&J\leq N,\\
x^0,&J>N.
\end{cases}
\]
Thus, when $N\geq T$, the output is a uniformly selected iterate among $x^0,\ldots,x^T$. All $N$ oracle calls are counted, including calls after the selected iterate. The budget may depend on the public parameter tuple.

Writing $\mathbb{P}$ for the joint law of the independent index $J$ and sample sequence, success requires that $\bar{x}$ be measurable, that $\|\nabla f(\bar{x})\|$ be integrable under $\mathbb{P}$, and that
\[
\int \|\nabla f(\bar{x}(u))\|\,d\mathbb{P}(u)\leq\varepsilon.
\]

\section*{Nonconvex Finite-Sum Gradient Optimization}
We consider minimizing $F(x)=n^{-1}\sum_{i=1}^n f_i(x)$ on a real Hilbert space $E$, where $n\geq1$ and the differentiable components $f_i:E\to\mathbb{R}$ may be nonconvex. Their gradients satisfy an averaged squared Lipschitz bound with constant $L>0$. The objective has a lower bound $F_{\mathrm{lb}}\in\mathbb{R}$ with $F(x^0)-F_{\mathrm{lb}}\leq\Delta$, where $x^0\in E$ and $\Delta\geq0$. A query $(i,x)$ returns the exact component gradient $\nabla f_i(x)$. Given $\varepsilon>0$, the goal is a measurable output $\bar{x}\in E$ with integrable squared gradient norm satisfying $\mathbb{E}\|\nabla F(\bar{x})\|^2\leq\varepsilon^2$, where expectation is over all algorithm and oracle randomness.

The optimization problem is
\begin{equation}
\label{eq:problem}
\inf_{x\in E} F(x),
\qquad F(x)=\frac{1}{n}\sum_{i=1}^n f_i(x).
\end{equation}
The space $E$ is equipped with a measurable structure, and its norm is induced by its inner product. The public inputs are $(E,n,L,\Delta,\varepsilon,x^0)$, with $n$ a positive integer. The component functions and oracle satisfy the following conditions.

\begin{assumption}[Differentiability and averaged smoothness]
Each $f_i$ is differentiable everywhere on $E$. For every $x,y\in E$,
\begin{equation}
\frac{1}{n}\sum_{i=1}^n
\|\nabla f_i(x)-\nabla f_i(y)\|^2
\leq L^2\|x-y\|^2.
\end{equation}
Consequently, the objective gradient is
\begin{equation}
\nabla F(x)=\frac{1}{n}\sum_{i=1}^n\nabla f_i(x).
\end{equation}
\end{assumption}

\begin{assumption}[Lower bound and initial gap]
There exists $F_{\mathrm{lb}}\in\mathbb{R}$ such that
\begin{align}
F_{\mathrm{lb}} &\leq F(x) \qquad \text{for every }x\in E,\\
F(x^0)-F_{\mathrm{lb}} &\leq\Delta.
\end{align}
The lower bound need not be attained.
\end{assumption}

\begin{assumption}[Exact component-gradient access]
At every call index, a query $(i,x)\in\{1,\ldots,n\}\times E$ returns $\nabla f_i(x)\in E$ for every realization of the oracle randomness. Each query costs one oracle call. Any internal algorithm randomness is independent of the oracle randomness.
\end{assumption}

The component functions are accessed through this oracle. Both the component index and the query point may be chosen adaptively. The algorithm is selected using only the public inputs; the witness $F_{\mathrm{lb}}$ is not supplied to it. An externally specified call budget $K\in\mathbb{N}$ may depend on the full public parameter tuple, and every component-gradient query counts toward this budget.

After $K$ calls, the algorithm returns $\bar{x}\in E$. Let $(\Omega,\mathcal{F},\mathbb{P})$ carry the joint law of the algorithm and oracle randomness. The output map $\bar{x}:\Omega\to E$ must be measurable, and $\omega\mapsto\|\nabla F(\bar{x}(\omega))\|^2$ must be integrable with respect to $\mathbb{P}$. The required accuracy is
\begin{equation}
\label{eq:accuracy}
\int_{\Omega}\|\nabla F(\bar{x}(\omega))\|^2\,d\mathbb{P}(\omega)
\leq\varepsilon^2.
\end{equation}

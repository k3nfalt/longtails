\section*{Reshuffled Smooth Convex Finite Sums}
We minimize $F(x)=n^{-1}\sum_{i=1}^n f_i(x)$ on a real Hilbert space $E$, where $n\geq1$ and the differentiable convex functions $f_i:E\to\mathbb{R}$ have gradient Lipschitz constants bounded individually by $L_{\max}>0$ and on average by $L_{\mathrm{avg}}\in(0,L_{\max}]$. Some global minimizer $x^\star$ satisfies $\|x^0-x^\star\|\leq R$ and $n^{-1}\sum_i\|\nabla f_i(x^\star)\|^2\leq\sigma^2$, for a given initial point $x^0\in E$ and bounds $R,\sigma\geq0$. Each oracle call returns one exact component gradient. Algorithms use epochwise gradient updates with random reshuffling. For $\varepsilon>0$, the measurable output $\bar{x}\in E$ must have integrable objective gap and satisfy $\mathbb{E}[F(\bar{x})-F(x^\star)]\leq\varepsilon$, with expectation over algorithm and oracle randomness.

The optimization problem is
\[
\min_{x\in E} F(x),
\qquad F(x)=\frac{1}{n}\sum_{i=1}^n f_i(x).
\]
The space $E$ carries its Borel sigma-algebra and may be infinite dimensional; its norm is induced by its inner product. All parameters $(E,n,L_{\mathrm{avg}},L_{\max},R,\sigma,\varepsilon,x^0)$ are public algorithm inputs, with ranges
\[
n\in\mathbb{N},\quad n\geq1,\qquad
0<L_{\mathrm{avg}}\leq L_{\max},\qquad
R\geq0,\quad \sigma\geq0,\quad \varepsilon>0.
\]
The component functions are accessed through the oracle under the following conditions.

\begin{assumption}[Component convexity and smoothness]
Each $f_i$ is differentiable everywhere on $E$. There exist constants $\ell_1,\ldots,\ell_n$ such that
\[
0\leq\ell_i\leq L_{\max},\qquad
\frac{1}{n}\sum_{i=1}^n\ell_i\leq L_{\mathrm{avg}}.
\]
For every $i\in\{1,\ldots,n\}$, $x,y\in E$, and $t\in[0,1]$,
\begin{align*}
f_i((1-t)x+ty)&\leq(1-t)f_i(x)+tf_i(y),\\
\|\nabla f_i(x)-\nabla f_i(y)\|&\leq\ell_i\|x-y\|.
\end{align*}
The individual constants $\ell_i$ are not supplied to the algorithm.
\end{assumption}

\begin{assumption}[Attainment and component gradients]
There exists $x^\star\in E$ such that
\begin{align*}
F(x^\star)&\leq F(x)\qquad\text{for every }x\in E,\\
\|x^0-x^\star\|&\leq R,\\
\frac{1}{n}\sum_{i=1}^n\|\nabla f_i(x^\star)\|^2&\leq\sigma^2.
\end{align*}
\end{assumption}

\begin{assumption}[Exact component-gradient access]
A query consists of an index $i\in\{1,\ldots,n\}$ and a point $x\in E$. At every call index and for every realization of oracle randomness, the reply is exactly $\nabla f_i(x)\in E$. Each query costs one oracle call. Query points may lie anywhere in $E$. Algorithm randomness is independent of oracle randomness.
\end{assumption}

\begin{assumption}[Permitted reshuffled updates]
The algorithm selects an integer $m\geq0$ and a deterministic sequence of epoch step sizes $(\eta_j)_{j\geq0}$ with $\eta_j\in\mathbb{R}$, using the public inputs. For epochs $j<m$, the permutations $\pi_j$ of $\{1,\ldots,n\}$ are independent and uniformly distributed. For epochs $j\geq m$, $\pi_j$ is the identity permutation.

Starting from $x_0=x^0$, call $t\geq0$ uses the epoch and within-epoch position
\[
j=\left\lfloor\frac{t}{n}\right\rfloor,
\qquad r=t-jn+1,
\]
and performs
\begin{align*}
i_t&=\pi_j(r),\\
g_t&=\nabla f_{i_t}(x_t),\\
x_{t+1}&=x_t-\eta_j g_t.
\end{align*}
Thus each completed epoch queries every component once, at successive iterates, with a common step size within that epoch.
\end{assumption}

An externally specified budget $B\in\mathbb{N}$ counts component-gradient calls and may depend on the full public parameter tuple. Execution stops after $B$ calls and returns the last iterate $\bar{x}=x_B$; if $B$ is not divisible by $n$, the final epoch is incomplete. The number $m$ of randomly ordered epochs and the call budget $B$ are distinct quantities.

Let $\mathbb{P}$ denote the joint probability law of algorithm and oracle randomness, and write $\bar{x}(\omega)$ for the output at outcome $\omega$. Success requires that $\bar{x}$ be measurable and that, for some global minimizer $x^\star$, the function $\omega\mapsto F(\bar{x}(\omega))-F(x^\star)$ be integrable with respect to $\mathbb{P}$ and satisfy
\[
\int \bigl(F(\bar{x}(\omega))-F(x^\star)\bigr)
\,\mathrm{d}\mathbb{P}(\omega)\leq\varepsilon.
\]
Since every global minimizer has the same objective value, the minimizer witnessing the initial distance and component-gradient bounds may be used in this criterion.

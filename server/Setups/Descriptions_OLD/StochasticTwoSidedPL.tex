\section*{Stochastic Two-Sided PL Minimax Optimization}
We consider $\inf_{x\in E}\sup_{y\in F}f(x,y)$ for a possibly nonconvex--nonconcave function $f:E\times F\to\mathbb{R}$, where $E$ is a finite-dimensional real inner product space and $F$ is a separable real Hilbert space. The partial gradients are jointly Lipschitz, both variables satisfy Polyak--{\L}ojasiewicz conditions, and each inner supremum is attained. Given initial points $x^0\in E$ and $y^0\in F$ satisfying a prescribed initial bound, each oracle call returns one unbiased partial-gradient estimate with bounded variance and a fresh independent sample. For $\varepsilon>0$, the random output $(\bar{x},\bar{y})$ must satisfy $\mathbb{E}[v(\bar{x})-V+v(\bar{x})-f(\bar{x},\bar{y})]\leq\varepsilon$, where $v(x)=\sup_{y\in F}f(x,y)$ and $V=\inf_{x\in E}v(x)$.

The minimax problem is
\[
\inf_{x\in E}\sup_{y\in F} f(x,y).
\]
The component norms are induced by their inner products, and the product norm is $\|(x,y)\|=\max\{\|x\|,\|y\|\}$. The public data consist of $E$, $F$, a measurable sample space $\Omega$, the initial points $(x^0,y^0)$, and constants $L,\mu_x,\mu_y,\sigma,\Delta,\varepsilon$ satisfying
\[
0<\mu_x\leq L,\qquad 0<\mu_y\leq L,\qquad
\sigma\geq0,\qquad \Delta\geq0,\qquad \varepsilon>0.
\]
The objective and the oracle distribution are accessed through the oracle described below. Admissible instances satisfy the following conditions.

\begin{assumption}[Partial-gradient smoothness]
The partial gradients $g_x(x,y)=\nabla_x f(x,y)\in E$ and $g_y(x,y)=\nabla_y f(x,y)\in F$ exist everywhere. For all $(x,y),(x',y')\in E\times F$,
\begin{align*}
\|g_x(x,y)-g_x(x',y')\|
&\leq L\bigl(\|x-x'\|+\|y-y'\|\bigr),\\
\|g_y(x,y)-g_y(x',y')\|
&\leq L\bigl(\|x-x'\|+\|y-y'\|\bigr).
\end{align*}
\end{assumption}

\begin{assumption}[Two-sided PL conditions and inner attainment]
For every $x,x'\in E$ and $y,y'\in F$,
\begin{align*}
2\mu_x\bigl(f(x,y)-f(x',y)\bigr)
&\leq \|g_x(x,y)\|^2,\\
2\mu_y\bigl(f(x,y')-f(x,y)\bigr)
&\leq \|g_y(x,y)\|^2.
\end{align*}
For each $x\in E$, there exists $y_x\in F$ such that $f(x,y')\leq f(x,y_x)$ for every $y'\in F$; hence $v(x)=f(x,y_x)$.
\end{assumption}

\begin{assumption}[Independent stochastic partial-gradient access]
There is a probability measure $p$ on $\Omega$ and jointly measurable maps
\[
G_x:(E\times F)\times\Omega\to E,
\qquad
G_y:(E\times F)\times\Omega\to F.
\]
For every $z\in E\times F$ and $i\in\{x,y\}$, the map $G_i(z,\cdot)$ is Bochner integrable, its squared centered norm is integrable, and
\begin{align*}
\int_\Omega G_i(z,\omega)\,dp(\omega)
&=g_i(z),\\
\int_\Omega\|G_i(z,\omega)-g_i(z)\|^2\,dp(\omega)
&\leq\sigma^2.
\end{align*}
At call $k$, the algorithm selects a point $z\in E\times F$ and a component $i\in\{x,y\}$, and receives $G_i(z,\xi_k)$, where $(\xi_k)_{k\geq0}$ are independent with common law $p$. Each component query costs one oracle call. Queries may be adaptive and may lie anywhere in $E\times F$. Internal algorithm randomness is independent of the entire sample sequence.
\end{assumption}

\begin{assumption}[Initial potential bound]
The initial points satisfy
\[
2v(x^0)-V-f(x^0,y^0)
+\frac{\|g_x(x^0,y^0)\|^2}{L}
\leq\Delta.
\]
\end{assumption}

The allowed algorithms are restricted to stochastic smoothed alternating gradient descent--ascent. They choose $r,\tau_x,\tau_y,\beta\in\mathbb{R}$ and $n\in\mathbb{N}$ from the public data, with no further parameter restrictions. Initialize $z^0=x^0$. Each complete round $t\geq0$ consists of an $x$-component call followed by a $y$-component call and the updates
\begin{align*}
x^{t+1}
&=x^t-\tau_x\bigl(G_x((x^t,y^t),\xi_{2t})
+r(x^t-z^t)\bigr),\\
y^{t+1}
&=y^t+\tau_yG_y((x^{t+1},y^t),\xi_{2t+1}),\\
z^{t+1}
&=z^t+\beta(x^{t+1}-z^t).
\end{align*}
Thus each complete round costs two calls, with the second query evaluated at the updated $x$ coordinate.

An externally specified budget $K\in\mathbb{N}$ may depend on the full public parameter tuple. Independently of the oracle samples, draw $J$ uniformly from $\{0,\ldots,n\}$. After $K$ calls, the prescribed output is
\[
(\bar{x},\bar{y})=
\begin{cases}
(x^{n+J},y^{n+J}),
&1\leq n+J\leq\lfloor K/2\rfloor,\\
(x^0,y^0),&\text{otherwise}.
\end{cases}
\]
An unpaired final $x$-component call counts toward the budget and does not change this output.

Writing $\mathbb{P}$ for the joint law of all randomness, the output must be measurable as an $E\times F$-valued random variable. Its accuracy expression must be integrable under $\mathbb{P}$, and success requires
\[
\int\bigl(v(\bar{x})-V+v(\bar{x})-f(\bar{x},\bar{y})\bigr)
\,d\mathbb{P}\leq\varepsilon.
\]
This criterion controls the expected sum of the outer value error and the inner maximization error at the returned pair.

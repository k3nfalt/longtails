\section*{Generalized Smooth Convex Stochastic Optimization}
We consider minimizing a differentiable convex function $f:E\to\mathbb{R}$ on a separable real Hilbert space $E$. Its gradient is differentiable and satisfies $\|D\nabla f(x)\|_{\mathrm{op}}\leq L_0+L_1\|\nabla f(x)\|$ for every $x\in E$, where $D$ denotes the Fr\'echet derivative, $L_0>0$, and $L_1\geq0$. A given point $x^0\in E$ satisfies $\|x^0-x^\star\|\leq R$ and $f(x^0)-f(x^\star)\leq\Delta$ for some global minimizer $x^\star$, with $R,\Delta>0$. Each oracle call returns an unbiased stochastic gradient with variance and exponential-moment bounds at scale $\sigma\geq0$, using independent samples. Given $\varepsilon>0$ and $\delta\in(0,1)$, a measurable output $\bar{x}\in E$ must satisfy $\mathbb{P}(f(\bar{x})-f(x^\star)>\varepsilon)\leq\delta$, where $\mathbb{P}$ is the joint randomness law.

The optimization problem is
\[
\min_{x\in E} f(x),
\]
where the norm on $E$ is induced by its inner product and $E$ carries its Borel sigma-algebra. The public inputs are $E$, a measurable sample space $\Omega$, and $(L_0,L_1,R,\Delta,x^0,\sigma,\varepsilon,\delta)$. The objective is accessed through the oracle described below.

\begin{assumption}[Convexity and generalized smoothness]
The function $f$ is differentiable everywhere, its gradient is Fr\'echet differentiable, and, for every $x,y\in E$ and $t\in[0,1]$,
\begin{align*}
f((1-t)x+ty) &\leq (1-t)f(x)+tf(y),\\
\|D\nabla f(x)\|_{\mathrm{op}} &\leq L_0+L_1\|\nabla f(x)\|.
\end{align*}
Here the operator norm is taken for bounded linear maps from $E$ to $E$.
\end{assumption}

\begin{assumption}[Attainment and initialization]
There exists $x^\star\in E$ such that
\begin{align*}
f(x^\star) &\leq f(x) \qquad \text{for every }x\in E,\\
\|x^0-x^\star\| &\leq R,\\
f(x^0)-f(x^\star) &\leq\Delta.
\end{align*}
\end{assumption}

\begin{assumption}[Independent stochastic-gradient access]
There exist a probability measure $p$ on $\Omega$, a jointly measurable map $g:E\times\Omega\to E$, and a probability law $P$ on $\Omega^{\mathbb{N}}$. Under $P$, the coordinate samples $(\xi_k)_{k\geq0}$ are measurable, mutually independent, and each has law $p$. At call $k$, a query $x\in E$ returns $g(x,\xi_k)\in E$ and costs one oracle call. Queries may be chosen adaptively anywhere in $E$. Any internal algorithm randomness is independent of the entire sample sequence.
\end{assumption}

\begin{assumption}[Unbiasedness and light tails]
For every fixed $x\in E$, the map $g(x,\cdot)$ is Bochner integrable under $p$. The scalar maps
\[
\omega\longmapsto\|g(x,\omega)-\nabla f(x)\|^2,
\qquad
\omega\longmapsto\exp\!\left(
\frac{\|g(x,\omega)-\nabla f(x)\|^2}{\sigma^2}
\right)
\]
are also integrable under $p$, and
\begin{align*}
\int_\Omega g(x,\omega)\,p(d\omega)
&=\nabla f(x),\\
\int_\Omega\|g(x,\omega)-\nabla f(x)\|^2\,p(d\omega)
&\leq\sigma^2,\\
\int_\Omega\exp\!\left(
\frac{\|g(x,\omega)-\nabla f(x)\|^2}{\sigma^2}
\right)\,p(d\omega)
&\leq e.
\end{align*}
When $\sigma=0$, the quotient in the exponential is defined to be zero, including when its numerator is nonzero.
\end{assumption}

The algorithm is selected using only the public inputs; $f$, $g$, $p$, and $P$ are not supplied to it. An externally specified budget $K\in\mathbb{N}$ may depend on the full public parameter tuple and counts every oracle call. After $K$ calls, the algorithm returns a measurable point $\bar{x}\in E$. Under the joint law $\mathbb{P}$ of its internal randomness and the oracle samples, the required accuracy is
\[
\mathbb{P}\bigl(f(\bar{x})-f(x^\star)>\varepsilon\bigr)
\leq\delta.
\]
The minimizer witnessing the initialization bounds may be used here, since all global minimizers have the same objective value.

\section*{Similar Finite-Sum Quasar Optimization}
We consider minimizing $F(x)=n^{-1}\sum_{i=1}^{n}f_i(x)$ on a real Hilbert space $E$, where $n\geq1$ and the functions $f_i:E\to\mathbb{R}$ are differentiable. The potentially nonconvex objective has an $L$-Lipschitz gradient, satisfies a quadratic lower curvature bound, and is strongly quasar-convex with parameters $\gamma\in(0,1]$ and $\mu\in(0,L]$ relative to a global minimizer $x^\star$. Component-gradient deviations from $\nabla F$ are $\delta$-Lipschitz, with $\delta\geq0$ and $384\delta^2\leq\gamma^3\mu^2$. A prescribed point $x^0\in E$ satisfies $F(x^0)-F(x^\star)\leq\Delta$, where $\Delta\geq0$, and quadratic growth holds about $x^\star$. Each oracle call returns one exact component gradient. For $\varepsilon>0$, the required output $\bar{x}\in E$ satisfies $F(\bar{x})-F(x^\star)\leq\varepsilon$ almost surely.

The optimization problem is
\[
\min_{x\in E} F(x),
\qquad F(x)=\frac{1}{n}\sum_{i=1}^{n}f_i(x),
\]
where the norm on $E$ is induced by its inner product. The public inputs are $E$, $L$, $x^0$, $n$, $\mu$, $\gamma$, $\delta$, $\Delta$, and $\varepsilon$, with the ranges and parameter restriction stated above. The component functions are accessed through the oracle. The admissible instances satisfy the following conditions.

\begin{assumption}[Smoothness and component similarity]
Each $f_i$ is differentiable everywhere on $E$. For every $x,y\in E$ and $i\in\{1,\ldots,n\}$,
\[
\begin{aligned}
\|\nabla F(x)-\nabla F(y)\|
&\leq L\|x-y\|,\\
\| (\nabla f_i(x)-\nabla F(x))
- (\nabla f_i(y)-\nabla F(y)) \|
&\leq\delta\|x-y\|.
\end{aligned}
\]
\end{assumption}

\begin{assumption}[Lower curvature bound]
For every $x,y\in E$,
\[
\begin{aligned}
F(y)\geq {}& F(x)+\langle\nabla F(x),y-x\rangle\\
&-\frac{\gamma\mu}{2\sqrt{\mu/L}}\|y-x\|^2.
\end{aligned}
\]
\end{assumption}

\begin{assumption}[Attainment, quasar convexity, and growth]
There exists a common point $x^\star\in E$ such that
\[
F(x^\star)\leq F(x)\quad\text{for every }x\in E,
\qquad F(x^0)-F(x^\star)\leq\Delta.
\]
For this same point and every $x\in E$,
\[
\begin{aligned}
\gamma F(x)+\langle\nabla F(x),x^\star-x\rangle
+\frac{\gamma\mu}{2}\|x^\star-x\|^2
&\leq\gamma F(x^\star),\\
\frac{\gamma\mu}{2(2-\gamma)}\|x-x^\star\|^2
&\leq F(x)-F(x^\star).
\end{aligned}
\]
\end{assumption}

\begin{assumption}[Exact component-gradient access]
At every call index, a query $(i,x)\in\{1,\ldots,n\}\times E$ returns the vector $\nabla f_i(x)\in E$ for every realization of the oracle randomness. Each such query costs one oracle call. Queries may be adaptive, and any internal algorithm randomness is independent of the oracle randomness.
\end{assumption}

The algorithm is selected using the public inputs. An externally specified budget $K\in\mathbb{N}$ may depend on the entire public parameter tuple. After $K$ oracle calls, the algorithm returns $\bar{x}\in E$. Writing $\mathbb{P}$ for the joint law of the algorithm and oracle randomness, success requires a global minimizer for which
\[
F(\bar{x})-F(x^\star)\leq\varepsilon
\qquad\mathbb{P}\text{-almost surely}.
\]
The minimizer appearing in the assumptions may be used here because all global minimizers have the same objective value.

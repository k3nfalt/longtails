\section*{Heavy-Tailed Polyak--Lojasiewicz Optimization}
We consider minimizing a differentiable function $f:E\to\mathbb{R}$ on a separable real Hilbert space $E$. Its gradient is $L$-Lipschitz, and $2\mu(f(x)-f_{\inf})\leq\|\nabla f(x)\|^2$ for every $x\in E$, where $0<\mu\leq L$ and $f_{\inf}=\inf_{y\in E}f(y)>-\infty$. Each oracle call returns an unbiased random gradient with uniformly bounded second and $q$th centered moments, where $q>2$, using independent samples across calls. From a given point $x^0\in E$, the admissible method is gradient descent with a positive deterministic step-size schedule. Its measurable last iterate $\bar{x}$ must satisfy $\mathbb{P}(f(\bar{x})-f_{\inf}>\varepsilon)\leq\delta$, where $\varepsilon>0$, $\delta\in(0,1)$, and $\mathbb{P}$ is the execution law.

The optimization problem is
\[
\inf_{x\in E} f(x),
\]
with the norm induced by the inner product on $E$. Equip $E$ with its Borel sigma-algebra. The setup parameters satisfy
\[
0<\mu\leq L,\qquad \sigma>0,\qquad \nu>0,\qquad q>2,
\qquad \Delta\geq0,\qquad \varepsilon>0,\qquad 0<\delta<1.
\]
The admissible objectives and oracle access satisfy the following conditions.

\begin{assumption}[Smoothness and Polyak--Lojasiewicz condition]
The function $f$ is differentiable everywhere, is bounded below, and satisfies, for every $x,y\in E$,
\begin{align*}
\|\nabla f(x)-\nabla f(y)\| &\leq L\|x-y\|,\\
2\mu\bigl(f(x)-f_{\inf}\bigr) &\leq \|\nabla f(x)\|^2.
\end{align*}
\end{assumption}

\begin{assumption}[Initial objective gap]
The given initial point satisfies
\[
f(x^0)-f_{\inf}\leq\Delta.
\]
\end{assumption}

\begin{assumption}[Independent stochastic-gradient access]
Let $\Omega$ be a measurable sample space, let $P_0$ be a probability measure on $\Omega$, and let $P$ be a probability measure on $\Omega^{\mathbb{N}}$. The coordinate samples $\xi_t$, $t=0,1,\ldots$, are measurable, mutually independent under $P$, and each has law $P_0$. Oracle $0$, the sole oracle, returns $g(x,\xi_t)\in E$ at call $t$ and query $x\in E$, where $g:E\times\Omega\to E$ is jointly measurable. Each reply costs one oracle call. The sample process is independent of any algorithm randomness.
\end{assumption}

\begin{assumption}[Unbiasedness and heavy-tail moment bounds]
For every $x\in E$, the map $g(x,\cdot)$ is Bochner integrable under $P_0$ and
\[
\int_{\Omega} g(x,\omega)\,dP_0(\omega)=\nabla f(x).
\]
The scalar maps $\omega\mapsto\|g(x,\omega)-\nabla f(x)\|^2$ and $\omega\mapsto\|g(x,\omega)-\nabla f(x)\|^q$ are integrable under $P_0$, with
\begin{align*}
\int_{\Omega}\|g(x,\omega)-\nabla f(x)\|^2\,dP_0(\omega)
&\leq\sigma^2,\\
\int_{\Omega}\|g(x,\omega)-\nabla f(x)\|^q\,dP_0(\omega)
&\leq\nu^q.
\end{align*}
\end{assumption}

Only $(E,L,\mu,x^0)$ are public algorithm inputs. The quantities $\sigma$, $\nu$, $q$, $\Delta$, $\varepsilon$, and $\delta$, as well as the sample space and its laws, are not supplied to the algorithm. An externally specified call budget $K\in\mathbb{N}$ may depend on the full setup parameters.

The admissible algorithm uses a deterministic schedule $(\eta_t)_{t\geq0}$ selected solely from the public inputs, with $\eta_t>0$ for every $t$. It queries oracle $0$ at its current iterate and updates according to
\[
x^{t+1}=x^t-\eta_t g(x^t,\xi_t),\qquad t=0,\ldots,K-1.
\]
Queries lie in $E$. After exactly $K$ calls, the output is the last iterate $\bar{x}=x^K$; for $K=0$, it is $x^0$. Success requires that $\bar{x}$ be measurable and that
\[
\mathbb{P}\bigl(f(\bar{x})-f_{\inf}>\varepsilon\bigr)\leq\delta,
\]
where $\mathbb{P}$ denotes the law induced by the oracle samples and the algorithm.
